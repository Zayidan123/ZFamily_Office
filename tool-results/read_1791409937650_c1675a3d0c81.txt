   256→  // ---------- Query settings ----------

   258→    queryKey: ["settings"],

   262→  const putM = useMutation({
   263→    mutationFn: (body: Record<string, unknown>) =>
   264→      apiFetch("/settings", { method: "PUT", body }),
   265→    onSuccess: () => {
   266→      toast.success("Pengaturan disimpan");
   267→      void qc.invalidateQueries({ queryKey: ["settings"] });
   268→      void qc.invalidateQueries({ queryKey: ["risk"] });

   270→    onError: (e) => toast.error("Gagal menyimpan pengaturan", { description: errMsg(e) }),
   271→  });
   272→
   273→  // ---------- Status sistem (panic mode) ----------
   274→  const statusQ = useQuery({
   275→    queryKey: ["system-status"],
   276→    queryFn: () => apiFetch<{ panicMode: boolean; version: string; serverTime: string }>("/system/status"),
   277→    refetchInterval: 15_000,
   278→    staleTime: 10_000,
   279→  });
   280→  const panicOn = statusQ.data?.panicMode === true;
   281→

   283→    mutationFn: (enabled: boolean) =>
   284→      apiFetch("/system/panic", { method: "POST", body: { enabled } }),

   286→      toast.success(enabled ? "PANIC MODE AKTIF" : "Panic mode dimatikan", {
   287→        description: enabled

   289→          : "Sistem kembali normal.",
   290→      });
   291→      setPanicOpen(false);
   292→      setPanicText("");
   293→      void qc.invalidateQueries({ queryKey: ["system-status"] });
   294→    },
   295→    onError: (e) => toast.error("Gagal mengubah panic mode", { description: errMsg(e) }),
   296→  });
   297→
   298→  const [panicOpen, setPanicOpen] = React.useState(false);
   299→  const [panicText, setPanicText] = React.useState("");
   300→
   301→  // ---------- Target alokasi ----------
   302→  const [targets, setTargets] = React.useState<AllocationTarget>(DEFAULT_TARGETS);
   303→  const [dirty, setDirty] = React.useState(false);
   304→
   305→  React.useEffect(() => {
   306→    const at = settingsQ.data?.allocationTargets;
   307→    if (at && !dirty) {
   308→      setTargets({
   309→        crypto: numOr(at.crypto, DEFAULT_TARGETS.crypto),
   310→        stocks: numOr(at.stocks, DEFAULT_TARGETS.stocks),
   311→        forexGold: numOr(at.forexGold, DEFAULT_TARGETS.forexGold),
   312→        property: numOr(at.property, DEFAULT_TARGETS.property),
   313→        cash: numOr(at.cash, DEFAULT_TARGETS.cash),
   314→      });
   315→    }
   316→  }, [settingsQ.data, dirty]);
   317→
   318→  const total =
   319→    targets.crypto + targets.stocks + targets.forexGold + targets.property + targets.cash;
   320→  const totalOk = Math.round(total) === 100;
   321→
   322→  const saveTargets = () => {
   323→    putM.mutate({ allocationTargets: targets });
   324→    setDirty(false);
   325→  };
   326→
   327→  // ---------- Ekspor data ----------
   328→  const exportM = useMutation({
   329→    mutationFn: async () => {
   330→      const data = await apiFetch<unknown>("/admin/export");
   331→      const now = new Date();
   332→      const stamp = `${now.getFullYear()}${String(now.getMonth() + 1).padStart(2, "0")}${String(
   333→        now.getDate(),
   334→      ).padStart(2, "0")}`;
   335→      const blob = new Blob([JSON.stringify(data, null, 2)], { type: "application/json" });
   336→      const url = URL.createObjectURL(blob);
   337→      const a = document.createElement("a");
   338→      a.href = url;
   339→      a.download = `family-office-export-${stamp}.json`;
   340→      document.body.appendChild(a);
   341→      a.click();
   342→      a.remove();
   343→      URL.revokeObjectURL(url);
   344→      return blob.size;
   345→    },
   346→    onSuccess: (size) =>
   347→      toast.success("Ekspor berhasil diunduh", {
   348→        description: `family-office-export-*.json · ${(size / 1024).toFixed(1)} KB`,
   349→      }),
   350→    onError: (e) => toast.error("Gagal mengekspor data", { description: errMsg(e) }),
   351→  });

   353→  // ---------- MFA — Autentikasi Dua Faktor (Task S-b) ----------

   367→    mutationFn: () => apiFetch<MfaSetup>("/auth/mfa/setup", { method: "POST" }),

   391→      toast.error("Gagal menyalin — salin manual dari teks secret");

   395→  const mfaEnableM = useMutation({
   396→    mutationFn: (code: string) =>
   397→      apiFetch<unknown>("/auth/mfa/enable", { method: "POST", body: { code } }),
   398→    onSuccess: () => {
   399→      toast.success("MFA diaktifkan", {
   400→        description: "Login berikutnya akan meminta kode authenticator.",
   401→      });
   402→      setSetupOpen(false);
   403→      setSetupCode("");
   404→      setSetupData(null);
   405→      void qc.invalidateQueries({ queryKey: ["auth", "mfa"] });

   407→    onError: (e) => toast.error("Kode ditolak", { description: errMsg(e) }),

   418→        description: "Login tidak lagi meminta kode authenticator.",
   419→      });
   420→      setDisableOpen(false);
   421→      setDisableCode("");
   422→      void qc.invalidateQueries({ queryKey: ["auth", "mfa"] });
   423→    },
   424→    onError: (e) => toast.error("Kode ditolak", { description: errMsg(e) }),

   447→            </Label>
   448→            <Select
   449→              value={displayCurrency}
   450→              onValueChange={(v) => {

   452→                if (isOwner) putM.mutate({ displayCurrency: v });

   455→              <SelectTrigger id="pref-currency" className="w-full sm:w-48">
   456→                <SelectValue />
   457→              </SelectTrigger>
   458→              <SelectContent>
   459→                <SelectItem value="USD">USD — Dolar AS ($)</SelectItem>
   460→                <SelectItem value="IDR">IDR — Rupiah (Rp)</SelectItem>

   462→            </Select>
   463→            <p className="text-xs text-muted-foreground">

   468→          </div>
   469→
   470→          <div className="space-y-2">
   471→            <Label htmlFor="pref-theme">

   473→                <Palette className="h-3.5 w-3.5" aria-hidden="true" />

   492→                <SelectItem value="system">

   510→            variant="outline"
   511→            className={cn(

   513→              totalOk

   517→          >

   521→        }
   522→      >
   523→        {settingsQ.isPending ? (
   524→          <LoadingRows rows={5} />
   525→        ) : (
   526→          <div className="space-y-5">
   527→            {ALLOC_KEYS.map(({ key, label }) => (
