   143→   143→function DeltaPct({ value, className }: { value: number | null | undefined; className?: string }) {
   144→   144→  if (value === null || value === undefined || !Number.isFinite(value)) {
   145→   145→    return <span className={cn("text-xs text-muted-foreground", className)}>—</span>;
   146→   146→  }
   147→   147→  const up = value >= 0;
   148→   148→  return (
   149→   149→    <span
   150→   150→      className={cn(
   151→   151→        "inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 text-xs font-medium tabular-nums",
   152→   152→        up
   153→   153→          ? "bg-emerald-500/10 text-emerald-600 dark:text-emerald-400"
   154→   154→          : "bg-rose-500/10 text-rose-600 dark:text-rose-400",
   155→   155→        className,
   156→   156→      )}
   157→   157→    >
   158→   158→      {up ? <TrendingUp className="h-3 w-3" aria-hidden="true" /> : <TrendingDown className="h-3 w-3" aria-hidden="true" />}
   159→   159→      {formatPct(value)}
   160→   160→    </span>
   161→   161→  );
   162→   162→}

   254→   254→function AssetFormDialog({
   255→   255→  open,
   256→   256→  onOpenChange,
   257→   257→  asset,
   258→   258→  entities,
   259→   259→}: {
   260→   260→  open: boolean;
   261→   261→  onOpenChange: (v: boolean) => void;
   262→   262→  asset: AssetRow | null;
   263→   263→  entities: Entity[];
   264→   264→}) {
   265→   265→  const queryClient = useQueryClient();
   266→   266→  const form = useForm<AssetFormValues>({
   267→   267→    resolver: zodResolver(assetFormSchema),
   268→   268→    defaultValues: EMPTY_ASSET_FORM,
   269→   269→  });
   270→   270→  const currency = useWatch({ control: form.control, name: "currency", defaultValue: "USD" });
   271→   271→
   272→   272→  React.useEffect(() => {
   273→   273→    if (!open) return;
   274→   274→    form.reset(
   275→   275→      asset
   276→   276→        ? {
   277→   277→            name: asset.name,
   278→   278→            assetClass: asset.assetClass,
   279→   279→            symbol: asset.symbol ?? "",
   280→   280→            entityId: String(asset.entityId),
   281→   281→            quantity: String(asset.quantity),
   282→   282→            unitPrice: String(asset.unitPrice),
   283→   283→            currency: asset.currency === "IDR" ? "IDR" : "USD",
   284→   284→            sourceType: SOURCE_META[asset.sourceType] ? asset.sourceType : "other",
   285→   285→            sourceLabel: asset.sourceLabel ?? "",
   286→   286→            notes: asset.notes ?? "",
   287→   287→          }
   288→   288→        : EMPTY_ASSET_FORM,
   289→   289→    );
   290→   290→  }, [open, asset, form]);
   291→   291→
   292→   292→  const saveMutation = useMutation({
   293→   293→    mutationFn: (values: AssetFormValues) => {
   294→   294→      const body = {
   295→   295→        name: values.name.trim(),
   296→   296→        assetClass: values.assetClass,
   297→   297→        symbol: values.symbol.trim() ? values.symbol.trim() : null,
   298→   298→        entityId: Number(values.entityId),
   299→   299→        quantity: Number(values.quantity),
   300→   300→        unitPrice: Number(values.unitPrice),
   301→   301→        currency: values.currency,
   302→   302→        sourceType: values.sourceType,
   303→   303→        sourceLabel: values.sourceLabel.trim(),
   304→   304→        notes: values.notes.trim(),
   305→   305→      };
   306→   306→      if (asset) {
   307→   307→        return apiFetch<AssetRow>(`/assets/${asset.id}`, { method: "PUT", body });
   308→   308→      }
   309→   309→      return apiFetch<AssetRow>("/assets", { method: "POST", body });
   310→   310→    },
   311→   311→    onSuccess: () => {
   312→   312→      toast.success(asset ? "Aset berhasil diperbarui" : "Aset berhasil ditambahkan", {
   313→   313→        description: asset ? asset.name : "Data aset baru tersimpan ke portofolio.",
   314→   314→      });
   315→   315→      void queryClient.invalidateQueries({ queryKey: ["assets"] });
   316→   316→      void queryClient.invalidateQueries({ queryKey: ["assets", "classes"] });
   317→   317→      void queryClient.invalidateQueries({ queryKey: ["dashboard"] });
   318→   318→      onOpenChange(false);
   319→   319→    },
   320→   320→    onError: (err) => {
   321→   321→      toast.error(asset ? "Gagal memperbarui aset" : "Gagal menambahkan aset", {
   322→   322→        description: errMsg(err),
   323→   323→      });
   324→   324→    },
   325→   325→  });
   326→   326→
   327→   327→  const onSubmit = form.handleSubmit((values) => saveMutation.mutate(values));
   328→   328→
   329→   329→  return (
   330→   330→    <Dialog open={open} onOpenChange={onOpenChange}>
   331→   331→      <DialogContent className="max-h-[90vh] overflow-y-auto scrollbar-thin sm:max-w-xl">
   332→   332→        <DialogHeader>
   333→   333→          <DialogTitle>{asset ? "Edit Aset" : "Tambah Aset"}</DialogTitle>
   334→   334→          <DialogDescription>
   335→   335→            {asset
   336→   336→              ? "Perbarui detail aset. Harga pasar live tetap dihitung otomatis oleh engine."
   337→   337→              : "Catat aset baru ke portofolio keluarga — kripto, saham, properti, dll."}
   338→   338→          </DialogDescription>
   339→   339→        </DialogHeader>
   340→   340→
   341→   341→        <form onSubmit={onSubmit} className="grid gap-4" noValidate>
   342→   342→          <div className="grid gap-1.5">
   343→   343→            <Label htmlFor="asset-name">Nama Aset *</Label>
   344→   344→            <Input
   345→   345→              id="asset-name"
   346→   346→              placeholder="cth. Bitcoin, Rumah Bandung, BBCA"
   347→   347→              {...form.register("name")}
   348→   348→            />
   349→   349→            <FieldError message={form.formState.errors.name?.message} />
   350→   350→          </div>

   366→   366→                          {c.label}
