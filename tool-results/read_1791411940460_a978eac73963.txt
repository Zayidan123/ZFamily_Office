   270→  const putM = useMutation({
   271→    mutationFn: (body: Record<string, unknown>) =>
   272→      apiFetch("/settings", { method: "PUT", body }),
   273→    onSuccess: () => {
   274→      toast.success("Pengaturan disimpan");
   275→      void qc.invalidateQueries({ queryKey: ["settings"] });
   276→      void qc.invalidateQueries({ queryKey: ["risk"] });

   279→  });
   280→
   281→  // ---------- Status sistem (panic mode) ----------
   282→  const statusQ = useQuery({
   283→    queryKey: ["system-status"],
   284→    queryFn: () => apiFetch<{ panicMode: boolean; version: string; serverTime: string }>("/system/status"),
   285→    refetchInterval: 15_000,
   286→    staleTime: 10_000,
   287→  });
   288→  const panicOn = statusQ.data?.panicMode === true;
   289→

   291→    mutationFn: (enabled: boolean) =>
   292→      apiFetch("/system/panic", { method: "POST", body: { enabled } }),

   294→      toast.success(enabled ? "PANIC MODE AKTIF" : "Panic mode dimatikan", {
   295→        description: enabled
   296→          ? "Semua operasi tulis backend diblokir (HTTP 423)."

   299→      setPanicOpen(false);
   300→      setPanicText("");
   301→      void qc.invalidateQueries({ queryKey: ["system-status"] });
   302→    },
   303→    onError: (e) => toast.error("Gagal mengubah panic mode", { description: errMsg(e) }),
   304→  });
   305→
   306→  const [panicOpen, setPanicOpen] = React.useState(false);
   307→  const [panicText, setPanicText] = React.useState("");
   308→
   309→  // ---------- Target alokasi ----------
   310→  const [targets, setTargets] = React.useState<AllocationTarget>(DEFAULT_TARGETS);
   311→  const [dirty, setDirty] = React.useState(false);
   312→
   313→  React.useEffect(() => {
   314→    const at = settingsQ.data?.allocationTargets;
   315→    if (at && !dirty) {
   316→      setTargets({
   317→        crypto: numOr(at.crypto, DEFAULT_TARGETS.crypto),
   318→        stocks: numOr(at.stocks, DEFAULT_TARGETS.stocks),
   319→        forexGold: numOr(at.forexGold, DEFAULT_TARGETS.forexGold),
   320→        property: numOr(at.property, DEFAULT_TARGETS.property),
   321→        cash: numOr(at.cash, DEFAULT_TARGETS.cash),
   322→      });
   323→    }
   324→  }, [settingsQ.data, dirty]);
   325→
   326→  const total =
   327→    targets.crypto + targets.stocks + targets.forexGold + targets.property + targets.cash;
   328→  const totalOk = Math.round(total) === 100;
   329→
   330→  const saveTargets = () => {

   333→  };

   337→    mutationFn: async () => {
   338→      const data = await apiFetch<unknown>("/admin/export");
   339→      const now = new Date();
   340→      const stamp = `${now.getFullYear()}${String(now.getMonth() + 1).padStart(2, "0")}${String(
   341→        now.getDate(),
   342→      ).padStart(2, "0")}`;
   343→      const blob = new Blob([JSON.stringify(data, null, 2)], { type: "application/json" });
   344→      const url = URL.createObjectURL(blob);
   345→      const a = document.createElement("a");
   346→      a.href = url;
   347→      a.download = `family-office-export-${stamp}.json`;
   348→      document.body.appendChild(a);
   349→      a.click();
   350→      a.remove();
   351→      URL.revokeObjectURL(url);
   352→      return blob.size;
   353→    },
   354→    onSuccess: (size) =>
   355→      toast.success("Ekspor berhasil diunduh", {
   356→        description: `family-office-export-*.json · ${(size / 1024).toFixed(1)} KB`,
   357→      }),
   358→    onError: (e) => toast.error("Gagal mengekspor data", { description: errMsg(e) }),
   359→  });

   375→        description: `${d.name} · ${(d.sizeBytes / 1024).toFixed(0)} KB · ${d.totalBackups} backup tersimpan`,
   376→      });
   377→    },
   378→    onError: (e) =>
   379→      toast.error("Gagal membuat backup", {

   381→      }),
   382→  });
   383→
   384→  const downloadBackupM = useMutation({
   385→    mutationFn: async (name: string) => {
   386→      const blob = await apiGetBlob(`/system/backups/${encodeURIComponent(name)}/download`);
   387→      const url = URL.createObjectURL(blob);
   388→      const a = document.createElement("a");
   389→      a.href = url;
   390→      a.download = name;
   391→      document.body.appendChild(a);
   392→      a.click();
   393→      a.remove();
   394→      URL.revokeObjectURL(url);
   395→      return blob.size;
   396→    },
   397→    onSuccess: (size, name) =>
   398→      toast.success("Backup diunduh", { description: `${name} · ${(size / 1024).toFixed(0)} KB` }),
   399→    onError: (e) =>
   400→      toast.error("Gagal mengunduh backup", {
   401→        description: e instanceof ApiError ? e.message : "Backend belum terjangkau",
   402→      }),
   403→  });
   404→
   405→  // ---------- MFA — Autentikasi Dua Faktor (Task S-b) ----------
   406→  const mfaStatusQ = useQuery<MfaStatus, ApiError>({
   407→    queryKey: ["auth", "mfa", "status"],
   408→    queryFn: () => apiFetch<MfaStatus>("/auth/mfa/status"),
   409→    staleTime: 30_000,
   410→  });
   411→  const mfaEnabled = mfaStatusQ.data?.enabled === true;
   412→
   413→  const [setupOpen, setSetupOpen] = React.useState(false);
   414→  const [setupData, setSetupData] = React.useState<MfaSetup | null>(null);
   415→  const [setupCode, setSetupCode] = React.useState("");
   416→  const [copied, setCopied] = React.useState(false);
   417→
   418→  const mfaSetupM = useMutation({
   419→    mutationFn: () => apiFetch<MfaSetup>("/auth/mfa/setup", { method: "POST" }),
   420→    onSuccess: (d) => setSetupData(d),
   421→    onError: (e) => {
   422→      toast.error("Gagal menyiapkan MFA", { description: errMsg(e) });
   423→      setSetupOpen(false);
   424→    },
   425→  });
   426→
   427→  const openSetup = () => {
   428→    setSetupData(null);
   429→    setSetupCode("");
   430→    setCopied(false);
   431→    setSetupOpen(true);
   432→    mfaSetupM.mutate();
   433→  };
   434→
   435→  const copySecret = async () => {
   436→    if (!setupData?.secret) return;
   437→    try {
   438→      await navigator.clipboard.writeText(setupData.secret);
   439→      setCopied(true);
   440→      toast.success("Secret disalin ke clipboard");
   441→      setTimeout(() => setCopied(false), 2000);
   442→    } catch {
   443→      toast.error("Gagal menyalin — salin manual dari teks secret");
   444→    }
   445→  };
   446→
   447→  const mfaEnableM = useMutation({
   448→    mutationFn: (code: string) =>
   449→      apiFetch<unknown>("/auth/mfa/enable", { method: "POST", body: { code } }),
   450→    onSuccess: () => {
   451→      toast.success("MFA diaktifkan", {
   452→        description: "Login berikutnya akan meminta kode authenticator.",
   453→      });
   454→      setSetupOpen(false);
   455→      setSetupCode("");
   456→      setSetupData(null);
   457→      void qc.invalidateQueries({ queryKey: ["auth", "mfa"] });
   458→    },
   459→    onError: (e) => toast.error("Kode ditolak", { description: errMsg(e) }),
   460→  });
   461→
   462→  const [disableOpen, setDisableOpen] = React.useState(false);
   463→  const [disableCode, setDisableCode] = React.useState("");
   464→
   465→  const mfaDisableM = useMutation({
   466→    mutationFn: (code: string) =>
   467→      apiFetch<unknown>("/auth/mfa/disable", { method: "POST", body: { code } }),
   468→    onSuccess: () => {
   469→      toast.success("MFA dinonaktifkan", {
   470→        description: "Login tidak lagi meminta kode authenticator.",
   471→      });
   472→      setDisableOpen(false);
   473→      setDisableCode("");
   474→      void qc.invalidateQueries({ queryKey: ["auth", "mfa"] });
   475→    },
   476→    onError: (e) => toast.error("Kode ditolak", { description: errMsg(e) }),
   477→  });
   478→
   479→  return (
   480→    <div className="space-y-6">
   481→      <PageHeader
   482→        title="Pengaturan"
   483→        subtitle="Preferensi tampilan, target alokasi, keamanan, dan portabilitas data."
   484→      />
   485→
   486→      {settingsQ.isError && (
   487→        <ErrorBox message={errMsg(settingsQ.error)} onRetry={() => void settingsQ.refetch()} />
   488→      )}
   489→
   490→      {/* ================= Preferensi ================= */}
   491→      <SectionCard title="Preferensi" description="Tampilan mata uang & tema aplikasi">
   492→        <div className="grid gap-6 sm:grid-cols-2">
   493→          <div className="space-y-2">
   494→            <Label htmlFor="pref-currency">
   495→              <span className="inline-flex items-center gap-1.5">
   496→                <DatabaseBackup className="h-3.5 w-3.5" aria-hidden="true" />
   497→                Mata Uang Tampilan
   498→              </span>
   499→            </Label>
   500→            <Select
   501→              value={displayCurrency}
   502→              onValueChange={(v) => {
   503→                setDisplayCurrency(v as "USD" | "IDR");
   504→                if (isOwner) putM.mutate({ displayCurrency: v });
   505→              }}
   506→            >
   507→              <SelectTrigger id="pref-currency" className="w-full sm:w-48">
   508→                <SelectValue />
   509→              </SelectTrigger>
   510→              <SelectContent>
   511→                <SelectItem value="USD">USD — Dolar AS ($)</SelectItem>
   512→                <SelectItem value="IDR">IDR — Rupiah (Rp)</SelectItem>
   513→              </SelectContent>
   514→            </Select>
   515→            <p className="text-xs text-muted-foreground">
   516→              {isOwner
   517→                ? "Tersimpan ke server & perangkat ini."
   518→                : "Tersimpan di perangkat ini (ubah server hanya owner)."}
   519→            </p>
   520→          </div>
   521→
   522→          <div className="space-y-2">
   523→            <Label htmlFor="pref-theme">
   524→              <span className="inline-flex items-center gap-1.5">
   525→                <Palette className="h-3.5 w-3.5" aria-hidden="true" />
   526→                Tema
   527→              </span>
   528→            </Label>
   529→            <Select value={theme ?? "system"} onValueChange={(v) => setTheme(v)}>
   530→              <SelectTrigger id="pref-theme" className="w-full sm:w-48">
   531→                <SelectValue />
   532→              </SelectTrigger>
   533→              <SelectContent>
