   121→function SideBadge({ side }: { side: Trade["side"] }) {
   122→  const isLong = side === "long" || side === "buy";

   137→/** Jumlah hari sejak tanggal ISO */
   138→function daysAgo(iso: string | null | undefined): number {
   139→  if (!iso) return Infinity;
   140→  const t = new Date(iso).getTime();
   141→  if (Number.isNaN(t)) return Infinity;
   142→  return Math.floor((Date.now() - t) / 86_400_000);
   143→}

   375→                        className={cn(
   376→                          "font-medium tabular-nums",

   423→                        {meta.label}
   424→                      </span>

   427→                </Card>

   508→        {/* Kontribusi */}

   554→      {/* ================= 4. Log Eksekusi ================= */}

   563→            <SelectContent>

   565→              {strategies.map((s) => (
   566→                <SelectItem key={s.id} value={s.id}>
   567→                  {s.name}
   568→                </SelectItem>
   569→              ))}
   570→            </SelectContent>
   571→          </Select>
   572→        }
   573→      >
   574→        {tradesQuery.isPending && strategyFilter !== "all" ? (
   575→          <LoadingRows rows={5} />
   576→        ) : (strategyFilter === "all" ? tradesAllQuery.isError : tradesQuery.isError) ? (
   577→          errBox(strategyFilter === "all" ? tradesAllQuery : tradesQuery)
   578→        ) : tableTrades.length === 0 ? (
   579→          <EmptyState
   580→            icon={<Activity className="h-6 w-6" />}
   581→            title="Belum ada trade"
   582→            description="Trade yang dieksekusi strategi akan tercatat di sini."
   583→          />
   584→        ) : (
   585→          <div className="max-h-96 overflow-y-auto rounded-lg border">
   586→            <Table>
   587→              <TableHeader className="sticky top-0 bg-background/95 backdrop-blur">
   588→                <TableRow>
   589→                  <TableHead className="w-[150px]">Waktu Close</TableHead>
   590→                  <TableHead>Strategi</TableHead>
   591→                  <TableHead>Symbol</TableHead>
   592→                  <TableHead>Side</TableHead>
   593→                  <TableHead className="text-right">Qty</TableHead>
   594→                  <TableHead className="text-right">Entry → Exit</TableHead>
   595→                  <TableHead className="text-right">PnL</TableHead>
   596→                </TableRow>
   597→              </TableHeader>
   598→              <TableBody>
   599→                {tableTrades.map((t) => (
   600→                  <TableRow key={t.id}>
   601→                    <TableCell className="tabular-nums">
   602→                      <div className="font-medium">
   603→                        {t.closedAt ? formatDateTime(t.closedAt) : "Terbuka"}
   604→                      </div>
   605→                      <div className="text-xs text-muted-foreground">
   606→                        {t.closedAt
   607→                          ? daysAgo(t.closedAt) < 30
   608→                            ? timeAgo(t.closedAt)
   609→                            : ""
   610→                          : `dibuka ${timeAgo(t.openedAt)}`}
   611→                      </div>
   612→                    </TableCell>
   613→                    <TableCell className="max-w-[160px] truncate font-medium">
   614→                      {t.strategyName}
   615→                    </TableCell>
   616→                    <TableCell className="font-mono text-xs font-semibold">{t.symbol}</TableCell>
   617→                    <TableCell>
   618→                      <SideBadge side={t.side} />
   619→                    </TableCell>

   621→                      {formatNumber(t.quantity, 4)}

   623→                    <TableCell className="text-right tabular-nums text-xs">
   624→                      {formatCurrency(t.entryPrice)}
   625→                      <span className="mx-1 text-muted-foreground">→</span>

   629→                      className={cn(
   630→                        "text-right font-semibold tabular-nums",
   631→                        t.pnlUsd > 0 && "text-emerald-600 dark:text-emerald-400",
   632→                        t.pnlUsd < 0 && "text-rose-600 dark:text-rose-400",
   633→                      )}
   634→                    >
   635→                      {t.pnlUsd > 0 ? "+" : ""}
   636→                      {formatCurrency(t.pnlUsd)}
   637→                    </TableCell>
   638→                  </TableRow>
   639→                ))}
   640→              </TableBody>
   641→            </Table>
   642→          </div>
   643→        )}
   644→      </SectionCard>
   645→
   646→      {/* ================= 5. Alert Sinyal Terpusat ================= */}
   647→      <SectionCard
   648→        title="Alert Sinyal Terpusat"
   649→        description="Notifikasi sinyal & eksekusi dari seluruh strategi"
   650→        actions={
   651→          <span className="flex items-center gap-1.5 text-xs text-muted-foreground">
   652→            <Bell className="h-4 w-4" aria-hidden="true" />
   653→            diperbarui tiap 60 detik
   654→          </span>
   655→        }
   656→      >
   657→        {alertsQuery.isPending ? (
   658→          <LoadingRows rows={4} />
   659→        ) : alertsQuery.isError ? (
   660→          errBox(alertsQuery)
   661→        ) : !alertsQuery.data || alertsQuery.data.length === 0 ? (
   662→          <EmptyState
   663→            icon={<Bell className="h-6 w-6" />}
   664→            title="Tidak ada alert"
   665→            description="Sinyal dan peringatan trading akan tampil di sini."
   666→          />
   667→        ) : (
   668→          <ul className="max-h-80 space-y-3 overflow-y-auto pr-1">
   669→            {alertsQuery.data.map((a) => (
   670→              <li key={a.id} className="flex items-start gap-3">
   671→                <AlertLevelBadge level={a.level} className="mt-0.5 shrink-0" />
   672→                <div className="min-w-0">
   673→                  <p className="text-sm leading-snug">{a.message}</p>
   674→                  <p className="mt-0.5 text-xs text-muted-foreground">

   676→                    {timeAgo(a.createdAt)}

   678→                </div>

   681→          </ul>

   683→      </SectionCard>
