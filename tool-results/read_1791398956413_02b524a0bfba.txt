   137→   137→/** Jumlah hari sejak tanggal ISO */
   138→   138→function daysAgo(iso: string | null | undefined): number {
   139→   139→  if (!iso) return Infinity;
   140→   140→  const t = new Date(iso).getTime();
   141→   141→  if (Number.isNaN(t)) return Infinity;
   142→   142→  return Math.floor((Date.now() - t) / 86_400_000);
   143→   143→}
