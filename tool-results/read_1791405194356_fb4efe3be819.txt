   128→// Sparkline mini — SVG polyline + area gradient
   129→// ------------------------------------------------------------
   130→function Sparkline({
   131→  values,
   132→  up,
   133→  className,
   134→}: {
   135→  values: number[];
   136→  up: boolean;
   137→  className?: string;
   138→}) {
   139→  const id = React.useId().replace(/[:]/g, "");
   140→  if (values.length < 2) return null;
   141→  const w = 120;
   142→  const h = 36;
   143→  const pad = 2;
   144→  const min = Math.min(...values);
   145→  const max = Math.max(...values);
   146→  const span = max - min || 1;
   147→  const step = (w - pad * 2) / (values.length - 1);
   148→  const coords = values.map((p, i) => {
   149→    const x = pad + i * step;
   150→    const y = h - pad - ((p - min) / span) * (h - pad * 2);
   151→    return [x, y] as const;
   152→  });
   153→  const linePath = coords
   154→    .map(([x, y], i) => `${i === 0 ? "M" : "L"}${x.toFixed(1)},${y.toFixed(1)}`)
   155→    .join(" ");
   156→  const last = coords[coords.length - 1];
   157→  const first = coords[0];
   158→  const areaPath = `${linePath} L${last[0].toFixed(1)},${h - pad} L${first[0].toFixed(1)},${h - pad} Z`;
   159→  const pointsAttr = coords.map(([x, y]) => `${x.toFixed(1)},${y.toFixed(1)}`).join(" ");
   160→  const color = up ? "#10b981" : "#f43f5e";
   161→
   162→  return (
   163→    <svg
   164→      viewBox={`0 0 ${w} ${h}`}
   165→      preserveAspectRatio="none"
   166→      className={cn("h-9 w-full", className)}
   167→      aria-hidden="true"
   168→    >
   169→      <defs>
   170→        <linearGradient id={`spark-${id}`} x1="0" y1="0" x2="0" y2="1">
   171→          <stop offset="0%" stopColor={color} stopOpacity={0.32} />
   172→          <stop offset="100%" stopColor={color} stopOpacity={0.02} />
   173→        </linearGradient>
   174→      </defs>
   175→      <path d={areaPath} fill={`url(#spark-${id})`} stroke="none" />
   176→      <polyline
   177→        points={pointsAttr}
   178→        fill="none"
   179→        stroke={color}
   180→        strokeWidth={1.8}
   181→        strokeLinejoin="round"
   182→        strokeLinecap="round"
   183→      />
   184→    </svg>
   185→  );
   186→}

   188→// ------------------------------------------------------------
   189→// Hook history per simbol
   190→// ------------------------------------------------------------
   191→function useMarketHistory(symbol: string, days: number, enabled = true) {
   192→  return useQuery<MarketHistory, ApiError>({
   193→    queryKey: ["market", "history", symbol, days],
   194→    queryFn: () =>
   195→      apiFetch<MarketHistory>("/market/history", { params: { symbol, days } }),
   196→    enabled: enabled && Boolean(symbol),
   197→    staleTime: 60_000,
   198→    refetchInterval: 60_000,
   199→  });
   200→}

   205→function WatchCard({
   206→  item,
   207→  isOwner,
   208→  onOpen,
   209→  onDelete,
   210→  deleting,
   211→}: {
   212→  item: WatchlistItem;
   213→  isOwner: boolean;
   214→  onOpen: () => void;
   215→  onDelete: () => void;
   216→  deleting: boolean;
   217→}) {
   218→  const historyQuery = useMarketHistory(item.symbol, 30);
   219→  const points = historyQuery.data?.points ?? [];
   220→  const values = React.useMemo(() => points.map((p) => p.price), [points]);
   221→  const up = item.price.change24hPct >= 0;
   222→
   223→  return (
   224→    <Card
   225→      role="button"
   226→      tabIndex={0}
   227→      aria-label={`Detail ${item.symbol}`}
   228→      onClick={onOpen}
   229→      onKeyDown={(e) => {
   230→        if (e.key === "Enter" || e.key === " ") {
   231→          e.preventDefault();
   232→          onOpen();
   233→        }
   234→      }}
   235→      className="group cursor-pointer rounded-xl p-5 transition-all duration-200 hover:-translate-y-0.5 hover:shadow-md focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
   236→    >
   237→      <div className="flex items-start justify-between gap-2">
   238→        <div className="min-w-0">
   239→          <p className="truncate text-lg font-bold tracking-tight">{item.symbol}</p>
   240→          {item.note ? (
   241→            <p className="mt-0.5 flex items-center gap-1 truncate text-xs text-muted-foreground">
   242→              <StickyNote className="h-3 w-3 shrink-0" aria-hidden="true" />
   243→              {item.note}
   244→            </p>
   245→          ) : (
   246→            <p className="mt-0.5 text-xs text-muted-foreground/60">
   247→              Ditambahkan {formatDate(item.addedAt)}
   248→            </p>
   249→          )}
   250→        </div>
   251→        {isOwner && (
   252→          <span onClick={(e) => e.stopPropagation()}>
   253→            <ConfirmButton
   254→              size="icon"
   255→              variant="ghost"
   256→              confirmLabel="?"
   257→              title={`Hapus ${item.symbol} dari watchlist`}
   258→              className="h-8 w-8 text-muted-foreground opacity-0 transition-opacity hover:text-rose-600 focus-visible:opacity-100 group-hover:opacity-100"
   259→              onConfirm={onDelete}
   260→              disabled={deleting}
   261→            >
   262→              {deleting ? (
   263→                <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />
   264→              ) : (
   265→                <Trash2 className="h-4 w-4" aria-hidden="true" />
   266→              )}
   267→            </ConfirmButton>
   268→          </span>
   269→        )}
   270→      </div>
   271→
   272→      <div className="mt-3 flex items-end justify-between gap-2">
   273→        <CurrencyText
   274→          usd={item.price.usd}
   275→          className="text-xl font-semibold tabular-nums"
   276→        />
   277→        <Badge
   278→          variant="outline"
   279→          className={cn(
   280→            "tabular-nums",
   281→            up
   282→              ? "border-emerald-500/40 bg-emerald-500/10 text-emerald-700 dark:text-emerald-400"
   283→              : "border-rose-500/40 bg-rose-500/10 text-rose-700 dark:text-rose-400",
   284→          )}
   285→        >
   286→          {formatPct(item.price.change24hPct)}
   287→        </Badge>
   288→      </div>
   289→
   290→      <div className="mt-2">
   291→        {historyQuery.isPending ? (
   292→          <Skeleton className="h-9 w-full rounded-md" />
   293→        ) : historyQuery.isError ? (
   294→          <p className="flex h-9 items-center text-xs text-muted-foreground">
   295→            Sparkline tidak tersedia
   296→          </p>
   297→        ) : (
   298→          <Sparkline values={values} up={up} />
   299→        )}
   300→      </div>
   301→    </Card>
   302→  );
   303→}

   308→function WatchDetailDialog({
   309→  item,
   310→  open,
   311→  onOpenChange,
   312→}: {
   313→  item: WatchlistItem | null;
   314→  open: boolean;
   315→  onOpenChange: (v: boolean) => void;
   316→}) {
   317→  const [days, setDays] = React.useState<7 | 30 | 90>(30);
   318→  const formatCurrency = useFormatCurrency();
   319→  const symbol = item?.symbol ?? "";
   320→
   321→  React.useEffect(() => {
   322→    if (open) setDays(30);
   323→  }, [open, symbol]);
   324→
   325→  const historyQuery = useMarketHistory(symbol, days, open && Boolean(item));
   326→
   327→  const points = historyQuery.data?.points ?? [];
   328→  const first = points.length > 0 ? points[0].price : null;
   329→  const lastPrice = points.length > 0 ? points[points.length - 1].price : null;
   330→  const periodChange =
   331→    first && lastPrice ? ((lastPrice - first) / first) * 100 : null;
   332→
   333→  return (
   334→    <Dialog open={open} onOpenChange={onOpenChange}>
   335→      <DialogContent className="sm:max-w-lg">
   336→        {item && (
   337→          <>
   338→            <DialogHeader>
   339→              <DialogTitle className="flex items-center gap-2 text-xl">
   340→                {item.symbol}
   341→                <Badge
   342→                  variant="outline"
   343→                  className={cn(
   344→                    "tabular-nums",
   345→                    item.price.change24hPct >= 0
   346→                      ? "border-emerald-500/40 bg-emerald-500/10 text-emerald-700 dark:text-emerald-400"
   347→                      : "border-rose-500/40 bg-rose-500/10 text-rose-700 dark:text-rose-400",
   348→                  )}
   349→                >
   350→                  {formatPct(item.price.change24hPct)} 24j
   351→                </Badge>
   352→              </DialogTitle>
   353→              <DialogDescription>
   354→                Harga sekarang{" "}
   355→                <CurrencyText
   356→                  usd={item.price.usd}
   357→                  className="font-semibold text-foreground"
   358→                />{" "}
   359→                · ditambahkan {formatDate(item.addedAt)}
   360→              </DialogDescription>
   361→            </DialogHeader>
   362→
   363→            <div className="flex items-center justify-between gap-3">
   364→              <p className="text-xs font-medium uppercase tracking-wide text-muted-foreground">
   365→                Riwayat Harga
   366→              </p>
   367→              <div className="flex gap-1" role="group" aria-label="Rentang hari">
   368→                {([7, 30, 90] as const).map((d) => (
   369→                  <Button
   370→                    key={d}
   371→                    type="button"
   372→                    size="sm"
   373→                    variant={days === d ? "default" : "outline"}
   374→                    className="h-7 px-2.5 text-xs"
   375→                    onClick={() => setDays(d)}
   376→                    aria-pressed={days === d}
   377→                  >
   378→                    {d} hari
   379→                  </Button>
   380→                ))}
   381→              </div>
   382→            </div>
   383→
   384→            <div className="h-[220px] w-full" role="img" aria-label={`Grafik harga ${symbol}`}>
   385→              {historyQuery.isPending ? (
   386→                <Skeleton className="h-full w-full rounded-lg" />
   387→              ) : historyQuery.isError ? (
   388→                <ErrorBox
   389→                  className="h-full"
   390→                  message={
   391→                    historyQuery.error instanceof ApiError
   392→                      ? historyQuery.error.message
   393→                      : undefined
   394→                  }
   395→                  onRetry={() => void historyQuery.refetch()}
   396→                />
   397→              ) : points.length === 0 ? (
   398→                <EmptyState title="Belum ada data riwayat" />
   399→              ) : (
   400→                <ResponsiveContainer width="100%" height="100%">
   401→                  <AreaChart data={points} margin={{ top: 8, right: 8, left: 0, bottom: 0 }}>
   402→                    <defs>
   403→                      <linearGradient id={`detailFill-${symbol}`} x1="0" y1="0" x2="0" y2="1">
   404→                        <stop offset="0%" stopColor="#10b981" stopOpacity={0.32} />
   405→                        <stop offset="100%" stopColor="#10b981" stopOpacity={0.02} />
   406→                      </linearGradient>
   407→                    </defs>
   408→                    <CartesianGrid
   409→                      strokeDasharray="3 3"
   410→                      stroke="rgba(113,113,122,0.18)"
   411→                      vertical={false}
   412→                    />
   413→                    <XAxis
   414→                      dataKey="date"
   415→                      tick={{ fill: "#71717a", fontSize: 10 }}
   416→                      tickLine={false}
   417→                      axisLine={false}
   418→                      minTickGap={32}
   419→                      tickFormatter={(v: string) => v.slice(5)}
   420→                    />
   421→                    <YAxis
   422→                      tickFormatter={(v: number) => formatCurrency(v, true)}
   423→                      tick={{ fill: "#71717a", fontSize: 10 }}
   424→                      tickLine={false}
   425→                      axisLine={false}
   426→                      width={70}
   427→                      domain={["auto", "auto"]}
   428→                    />
   429→                    <Tooltip
   430→                      cursor={{ stroke: "rgba(113,113,122,0.35)", strokeDasharray: "4 4" }}
   431→                      contentStyle={{
   432→                        background: "var(--popover)",

   463→                    "font-semibold tabular-nums",

   498→function AddWatchDialog({
   499→  open,
   500→  onOpenChange,
   501→  symbols,

   504→  onSubmit,

   564→            <Label htmlFor="watch-note">Catatan (opsional)</Label>

   569→              placeholder="mis. DCA bulanan, entry zone akumulasi…"

   571→            />
   572→          </div>
   573→        </div>
   574→
   575→        <DialogFooter>
   576→          <Button type="button" variant="outline" onClick={() => onOpenChange(false)}>
   577→            Batal
   578→          </Button>
   579→          <Button
   580→            type="button"
   581→            className="bg-emerald-600 text-white hover:bg-emerald-700"
   582→            disabled={!symbol || pending}
   583→            onClick={() => onSubmit(symbol, note.trim())}
   584→          >
   585→            {pending ? (
   586→              <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />
   587→            ) : (
   588→              <Plus className="h-4 w-4" aria-hidden="true" />
   589→            )}
   590→            Pantau
   591→          </Button>
   592→        </DialogFooter>
   593→      </DialogContent>
   594→    </Dialog>
   595→  );

   599→// Progress mini alert — % jarak harga kini menuju threshold (clamp 0-100)
   600→// above: current/threshold · below: threshold/current · triggered = 100
   601→// ------------------------------------------------------------
   602→function alertProgressPct(alert: PriceAlert): number {
   603→  if (alert.status === "triggered") return 100;
   604→  const cur = alert.currentPrice;
   605→  const th = alert.threshold;
   606→  if (!Number.isFinite(cur) || !Number.isFinite(th) || cur <= 0 || th <= 0) return 0;
   607→  const raw = alert.direction === "above" ? (cur / th) * 100 : (th / cur) * 100;
   608→  return Math.min(100, Math.max(0, raw));
   609→}

  1098→// Konverter Mata Uang (PRD #10) — pivot USD, rate live dari /rates
  1099→// Daftar: USD, IDR, BTC, ETH, BNB, SUI, XAU (per troy oz)
  1100→// ------------------------------------------------------------
  1101→const CONVERTER_CCYS = ["USD", "IDR", "BTC", "ETH", "BNB", "SUI", "XAU"] as const;
  1102→type ConverterCcy = (typeof CONVERTER_CCYS)[number];
  1103→
  1104→function fmtConverterValue(v: number | null, ccy: ConverterCcy): string {
  1105→  if (v === null || !Number.isFinite(v)) return "—";
  1106→  if (ccy === "IDR") return formatIdr(v);
  1107→  if (ccy === "USD") return formatUsd(v);
  1108→  const abs = Math.abs(v);
  1109→  const digits = abs >= 1000 ? 2 : abs >= 1 ? 4 : abs >= 0.01 ? 6 : 8;
  1110→  return `${new Intl.NumberFormat("id-ID", {
  1111→    maximumFractionDigits: digits,
  1112→  }).format(v)} ${ccy}`;
  1113→}
  1114→
  1115→function ConverterCard() {
  1116→  const ratesQuery = useRates();
  1117→  const rates = ratesQuery.data;
  1118→
  1119→  const [amountRaw, setAmountRaw] = React.useState("1");
  1120→  const [from, setFrom] = React.useState<ConverterCcy>("BTC");
  1121→  const [to, setTo] = React.useState<ConverterCcy>("IDR");
  1122→
  1123→  const amount = Number(amountRaw.replace(",", "."));
  1124→  const amountValid = amountRaw.trim() !== "" && Number.isFinite(amount) && amount >= 0;
  1125→
  1126→  const usdIdr = typeof rates?.USDIDR === "number" && rates.USDIDR > 0 ? rates.USDIDR : null;
  1127→
  1128→  const toUsd = React.useCallback(
  1129→    (ccy: ConverterCcy, value: number): number | null => {
  1130→      if (ccy === "USD") return value;
  1131→      if (ccy === "IDR") return usdIdr ? value / usdIdr : null;
  1132→      const p = rates?.prices?.[ccy]?.usd;
  1133→      return typeof p === "number" && p > 0 ? value * p : null;
  1134→    },
  1135→    [rates, usdIdr],

  1144→    },
  1145→    [rates, usdIdr],
  1146→  );
  1147→
  1148→  const usdAmount = amountValid ? toUsd(from, amount) : null;
  1149→  const result = usdAmount !== null ? fromUsd(to, usdAmount) : null;
