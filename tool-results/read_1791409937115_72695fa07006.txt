   186→const numberString = (label: string) =>
   187→  z
   188→    .string()
   189→    .min(1, `${label} wajib diisi`)
   190→    .refine((v) => !Number.isNaN(Number(v)), `${label} harus berupa angka`)

   222→function TxFormDialog({
   223→  open,
   224→  onOpenChange,
   225→  assets,
   226→}: {
   227→  open: boolean;
   228→  onOpenChange: (v: boolean) => void;
   229→  assets: Asset[];
   230→}) {
   231→  const queryClient = useQueryClient();
   232→  const form = useForm<TxFormValues>({
   233→    resolver: zodResolver(txFormSchema),
   234→    defaultValues: { ...EMPTY_TX_FORM, at: nowLocalInput() },
   235→  });
   236→
   237→  const typeWatch = useWatch({ control: form.control, name: "type", defaultValue: "" });
   238→  const assetIdWatch = useWatch({ control: form.control, name: "assetId", defaultValue: "" });
   239→  const type = (typeWatch ?? "") as TransactionType | "";
   240→  const selectedAsset = assets.find((a) => String(a.id) === assetIdWatch) ?? null;
   241→
   242→  React.useEffect(() => {
   243→    if (!open) return;
   244→    form.reset({ ...EMPTY_TX_FORM, at: nowLocalInput() });
   245→  }, [open, form]);
   246→
   247→  const createMutation = useMutation({
   248→    mutationFn: (values: TxFormValues) => {
   249→      const body: Record<string, unknown> = {
   250→        assetId: Number(values.assetId),
   251→        type: values.type,
   252→        quantity: Number(values.quantity),
   253→        price: Number(values.price),
   254→        fee: values.fee === "" ? 0 : Number(values.fee),
   255→        note: values.note.trim(),
   256→      };
   257→      if (values.at) {
   258→        const d = new Date(values.at);
   259→        if (!Number.isNaN(d.getTime())) body.at = d.toISOString();
   260→      }
   261→      return apiFetch<Transaction>("/transactions", { method: "POST", body });
   262→    },
   263→    onSuccess: (tx) => {
   264→      toast.success("Transaksi berhasil dicatat", {
   265→        description: `${txMeta(tx.type).label} · ${tx.assetName ?? "aset"} · ${formatNumber(tx.quantity, 6)}`,
   266→      });
   267→      void queryClient.invalidateQueries({ queryKey: ["transactions"] });
   268→      void queryClient.invalidateQueries({ queryKey: ["assets"] });
   269→      void queryClient.invalidateQueries({ queryKey: ["dashboard"] });
   270→      onOpenChange(false);
   271→    },
   272→    onError: (err) => {
   273→      toast.error("Gagal mencatat transaksi", { description: errMsg(err) });
   274→    },
   275→  });
   276→
   277→  const onSubmit = form.handleSubmit((values) => {
   278→    // Validasi client-side terkait tipe & ketersediaan aset
   279→    const t = values.type as TransactionType;
   280→    const qty = Number(values.quantity);
   281→    if (["buy", "sell", "transfer_in", "transfer_out"].includes(t) && qty <= 0) {
   282→      form.setError("quantity", { message: "Jumlah harus lebih dari 0 untuk jenis transaksi ini" });
   283→      return;
   284→    }
   285→    if (t === "sell" || t === "transfer_out") {
   286→      const asset = assets.find((a) => String(a.id) === values.assetId);
   287→      if (asset && qty > asset.quantity + 1e-9) {
   288→        form.setError("quantity", {
   289→          message: `Melebihi jumlah tersedia (${formatNumber(asset.quantity, 6)})`,
   290→        });
   291→        return;
   292→      }
   293→    }
   294→    createMutation.mutate(values);
   295→  });
   296→
   297→  const needsPositiveQty =
   298→    type === "buy" || type === "sell" || type === "transfer_in" || type === "transfer_out";
   299→
   300→  return (
   301→    <Dialog open={open} onOpenChange={onOpenChange}>
   302→      <DialogContent className="max-h-[90vh] overflow-y-auto scrollbar-thin sm:max-w-lg">
   303→        <DialogHeader>
   304→          <DialogTitle>Catat Transaksi</DialogTitle>
   305→          <DialogDescription>
   306→            Beli/jual otomatis menyesuaikan jumlah aset. Harga &amp; fee mengikuti mata uang aset.
   307→          </DialogDescription>
   308→        </DialogHeader>
   309→
   310→        <form onSubmit={onSubmit} className="grid gap-4" noValidate>
   311→          <div className="grid gap-1.5">
   312→            <Label>Aset *</Label>
   313→            <Controller
   314→              control={form.control}
   315→              name="assetId"
   316→              render={({ field }) => (
   317→                <Select value={field.value} onValueChange={field.onChange}>
   318→                  <SelectTrigger className="w-full">
   319→                    <SelectValue placeholder="Pilih aset" />
   320→                  </SelectTrigger>
   321→                  <SelectContent className="max-h-64">
   322→                    {assets.length === 0 ? (
   323→                      <div className="px-3 py-2 text-sm text-muted-foreground">
   324→                        Belum ada aset — tambahkan aset terlebih dahulu
   325→                      </div>
   326→                    ) : (
   327→                      assets.map((a) => (
   328→                        <SelectItem key={String(a.id)} value={String(a.id)}>
   329→                          {a.name}
   330→                          {a.symbol ? ` (${a.symbol})` : ""}
   331→                        </SelectItem>
   332→                      ))
   333→                    )}
   334→                  </SelectContent>
   335→                </Select>
   336→              )}
   337→            />
   338→            {selectedAsset && (
   339→              <div className="flex flex-wrap items-center gap-2 text-xs text-muted-foreground">
   340→                <ClassBadge assetClass={selectedAsset.assetClass} className="px-1.5 text-[10px]" />
   341→                <span>
   342→                  Mata uang: <span className="font-medium">{selectedAsset.currency}</span>
   343→                </span>
   344→                <span>
   345→                  Tersedia:{" "}
   346→                  <span className="font-medium tabular-nums">
   347→                    {formatNumber(selectedAsset.quantity, 6)}
   348→                    {selectedAsset.symbol ? ` ${selectedAsset.symbol}` : ""}
   349→                  </span>
   350→                </span>
   351→              </div>
   352→            )}
   353→            <FieldError message={form.formState.errors.assetId?.message} />
   354→          </div>
   355→
   356→          <div className="grid gap-1.5">
   357→            <Label>Jenis Transaksi *</Label>
   358→            <Controller
   359→              control={form.control}
   360→              name="type"
   361→              render={({ field }) => (
   362→                <Select value={field.value} onValueChange={field.onChange}>
   363→                  <SelectTrigger className="w-full">
   364→                    <SelectValue placeholder="Pilih jenis" />
   365→                  </SelectTrigger>
   366→                  <SelectContent>
   367→                    {TX_TYPE_OPTIONS.map((t) => (
   368→                      <SelectItem key={t.value} value={t.value}>
   369→                        {t.label}
   370→                      </SelectItem>
   371→                    ))}
   372→                  </SelectContent>
   373→                </Select>
   374→              )}
   375→            />
   376→            {type === "valuation_update" && (
   377→              <p className="flex items-start gap-1.5 text-xs text-muted-foreground">
   378→                <Info className="mt-0.5 h-3 w-3 shrink-0" aria-hidden="true" />
   379→                Harga baru akan diterapkan sebagai harga satuan aset.
   380→              </p>
   381→            )}
   382→            <FieldError message={form.formState.errors.type?.message} />
   383→          </div>
   384→
   385→          <div className="grid gap-4 sm:grid-cols-2">
   386→            <div className="grid gap-1.5">
   387→              <Label htmlFor="tx-quantity">Jumlah *</Label>
   388→              <Input
   389→                id="tx-quantity"
   390→                type="number"
   391→                step="any"
   392→                min="0"
   393→                inputMode="decimal"
   394→                placeholder={needsPositiveQty ? "cth. 0.25" : "cth. 0"}
   395→                {...form.register("quantity")}
   396→              />
   397→              <FieldError message={form.formState.errors.quantity?.message} />
   398→            </div>
   399→
   400→            <div className="grid gap-1.5">
   401→              <div className="flex items-center justify-between gap-2">
   402→                <Label htmlFor="tx-price">Harga Satuan *</Label>
   403→                {selectedAsset && (
   404→                  <span className="rounded-md border bg-muted px-1.5 py-0.5 text-[10px] font-semibold uppercase text-muted-foreground">
   405→                    {selectedAsset.currency}
   406→                  </span>
   407→                )}
   408→              </div>
   409→              <Input
   410→                id="tx-price"
   411→                type="number"
   412→                step="any"
   413→                min="0"
   414→                inputMode="decimal"

   516→    onError: (e) =>
   517→      toast.error("Gagal mengekspor CSV", {
   518→        description: e instanceof ApiError ? e.message : "Backend belum terjangkau — mencoba lagi",
   519→      }),
   520→  });
   521→
   522→  // ---------- queries ----------
   523→  const assetsQuery = useQuery<Asset[], ApiError>({
   524→    queryKey: ["assets", "options"],
   525→    queryFn: () => apiFetch<Asset[]>("/assets"),
   526→    staleTime: 30_000,
   527→  });
   528→  const assets = assetsQuery.data ?? [];
   529→
   530→  const txParams = {
   531→    assetId: assetFilter !== "all" ? assetFilter : undefined,
   532→    type: typeFilter !== "all" ? typeFilter : undefined,
   533→    limit: 200,
   534→  };
   535→  const txQuery = useQuery<Transaction[], ApiError>({
   536→    queryKey: ["transactions", { assetId: txParams.assetId ?? null, type: txParams.type ?? null }],
   537→    queryFn: () =>
   538→      apiFetch<Transaction[]>("/transactions", {
   539→        params: { assetId: txParams.assetId, type: txParams.type, limit: 200 },
   540→      }),
   541→    staleTime: 30_000,
   542→  });

   557→      else if (OUTFLOW_TYPES.includes(tx.type)) outflow += Math.abs(tx.totalUsd ?? 0);

   559→    return { inflow, outflow, count: txs.length };
   560→  }, [txs]);
   561→
   562→  return (
   563→    <div className="space-y-6">
   564→      <PageHeader
   565→        title="Transaksi"
   566→        subtitle="Riwayat buy/sell, transfer, income/expense, dan valuation update."

   568→
   569→      {/* ================= StatCards arus kas ================= */}
   570→      <div className="grid gap-4 sm:grid-cols-3" aria-busy={txQuery.isPending && txs.length === 0}>
   571→        <StatCard
   572→          loading={txQuery.isPending && txs.length === 0}
   573→          label="Total Masuk"

   579→          loading={txQuery.isPending && txs.length === 0}
   580→          label="Total Keluar"

   596→      {/* ================= Toolbar ================= */}
   597→      <div className="flex flex-col gap-3 lg:flex-row lg:items-center lg:justify-between">
   598→        <div className="flex flex-wrap items-center gap-3">

   650→                type="button"
   651→                onClick={() => setFormOpen(true)}
   652→                className="bg-emerald-600 text-primary-foreground hover:bg-emerald-700"

   654→                <Plus className="h-4 w-4" aria-hidden="true" />

   678→            <LoadingRows rows={6} />

   692→              action={

   695→                    type="button"

   697→                    className="bg-emerald-600 text-primary-foreground hover:bg-emerald-700"

   715→                    <TableHead className="w-[110px] text-right">Jumlah</TableHead>

   717→                    <TableHead className="w-[130px] text-right">Total</TableHead>

   729→                      <TableRow key={String(tx.id)}>
   730→                        <TableCell>
   731→                          <div className="flex flex-col">
   732→                            <span className="whitespace-nowrap text-sm tabular-nums">

   738→                        <TableCell>
   739→                          <span className="inline-flex items-center gap-1.5">
   740→                            <DirIcon
   741→                              className={cn(
   742→                                "h-3.5 w-3.5",
   743→                                neutral && "text-zinc-500",
   744→                                !neutral && inflow && "text-emerald-600 dark:text-emerald-400",
   745→                                !neutral && !inflow && "text-rose-600 dark:text-rose-400",
   746→                              )}
   747→                              aria-hidden="true"
   748→                            />
   749→                            <TypeBadge type={tx.type} />
   750→                          </span>
   751→                        </TableCell>
   752→                        <TableCell className="font-medium">{tx.assetName ?? `#${tx.assetId}`}</TableCell>
   753→                        <TableCell className="text-right tabular-nums">
   754→                          {formatNumber(tx.quantity, 6)}
   755→                        </TableCell>

   757→                          <MoneyCell usd={tx.priceUsd} />
   758→                        </TableCell>
   759→                        <TableCell
   760→                          className={cn(
   761→                            "text-right font-semibold tabular-nums",
   762→                            neutral && "text-muted-foreground",
   763→                            !neutral && inflow && "text-emerald-600 dark:text-emerald-400",
   764→                            !neutral && !inflow && "text-rose-600 dark:text-rose-400",
   765→                          )}
   766→                        >
   767→                          <MoneyCell
   768→                            usd={tx.totalUsd}
   769→                            sign={neutral ? undefined : inflow ? "+" : "−"}
   770→                            className={cn(
   771→                              "font-semibold",
   772→                              neutral && "text-muted-foreground",
   773→                              !neutral && inflow && "text-emerald-600 dark:text-emerald-400",
   774→                              !neutral && !inflow && "text-rose-600 dark:text-rose-400",
   775→                            )}
   776→                          />
   777→                        </TableCell>
   778→                        <TableCell className="text-right">
   779→                          {tx.feeUsd > 0 ? (
   780→                            <MoneyCell usd={tx.feeUsd} className="text-muted-foreground" />
   781→                          ) : (
   782→                            <span className="text-muted-foreground">—</span>
   783→                          )}
   784→                        </TableCell>
   785→                        <TableCell>
   786→                          {tx.note?.trim() ? (
   787→                            <Tooltip>
   788→                              <TooltipTrigger asChild>
   789→                                <span className="block max-w-[220px] cursor-default truncate text-sm text-muted-foreground">
   790→                                  {tx.note}
   791→                                </span>
   792→                              </TooltipTrigger>
   793→                              <TooltipContent side="top" className="max-w-xs">
   794→                                <p className="text-xs">{tx.note}</p>
   795→                              </TooltipContent>
   796→                            </Tooltip>
   797→                          ) : (
   798→                            <span className="text-muted-foreground">—</span>
   799→                          )}
   800→                        </TableCell>
   801→                      </TableRow>
   802→                    );
   803→                  })}
   804→                </TableBody>
   805→              </Table>
   806→            </div>
   807→          </TooltipProvider>
   808→        )}
   809→      </SectionCard>
   810→
   811→      {txs.length > 0 && (
   812→        <p className="text-xs text-muted-foreground">
   813→          Nilai ditampilkan dalam mata uang tampilan pilihan Anda. Total masuk/keluar dihitung dari
   814→          tipe: masuk (beli, transfer masuk, pemasukan) dan keluar (jual, transfer keluar, pengeluaran).
   815→        </p>
   816→      )}
   817→
   818→      {/* ================= Dialog ================= */}
   819→      {isOwner && (
   820→        <TxFormDialog open={formOpen} onOpenChange={setFormOpen} assets={assets} />
   821→      )}
   822→    </div>
   823→  );
