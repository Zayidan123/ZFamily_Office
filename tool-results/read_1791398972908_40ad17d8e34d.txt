   142→/** datetime-local value untuk "sekarang" (zona lokal) */
   143→function nowLocalInput(): string {
   144→  const d = new Date();
   145→  const pad = (n: number) => String(n).padStart(2, "0");
   146→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
   147→}

   185→const numberString = (label: string) =>
   186→  z
   187→    .string()
   188→    .min(1, `${label} wajib diisi`)
   189→    .refine((v) => !Number.isNaN(Number(v)), `${label} harus berupa angka`)
   190→    .refine((v) => Number(v) >= 0, `${label} tidak boleh negatif`);

   192→const txFormSchema = z.object({
   193→  assetId: z.string().min(1, "Pilih aset"),
   194→  type: z.string().min(1, "Pilih jenis transaksi"),
   195→  quantity: numberString("Jumlah"),
   196→  price: numberString("Harga"),
   197→  fee: z
   198→    .string()
   199→    .refine((v) => v === "" || !Number.isNaN(Number(v)), "Fee harus berupa angka")
   200→    .refine((v) => v === "" || Number(v) >= 0, "Fee tidak boleh negatif"),
   201→  at: z
   202→    .string()
   203→    .refine((v) => v === "" || !Number.isNaN(new Date(v).getTime()), "Tanggal tidak valid"),
   204→  note: z.string().max(1000, "Catatan maksimal 1000 karakter"),
   205→});

   221→function TxFormDialog({
   222→  open,
   223→  onOpenChange,
   224→  assets,
   225→}: {
   226→  open: boolean;
   227→  onOpenChange: (v: boolean) => void;
   228→  assets: Asset[];
   229→}) {
   230→  const queryClient = useQueryClient();
   231→  const form = useForm<TxFormValues>({
   232→    resolver: zodResolver(txFormSchema),
   233→    defaultValues: { ...EMPTY_TX_FORM, at: nowLocalInput() },
   234→  });
   235→
   236→  const typeWatch = useWatch({ control: form.control, name: "type", defaultValue: "" });
   237→  const assetIdWatch = useWatch({ control: form.control, name: "assetId", defaultValue: "" });
   238→  const type = (typeWatch ?? "") as TransactionType | "";
   239→  const selectedAsset = assets.find((a) => String(a.id) === assetIdWatch) ?? null;
   240→
   241→  React.useEffect(() => {
   242→    if (!open) return;
   243→    form.reset({ ...EMPTY_TX_FORM, at: nowLocalInput() });
   244→  }, [open, form]);
   245→
   246→  const createMutation = useMutation({
   247→    mutationFn: (values: TxFormValues) => {
   248→      const body: Record<string, unknown> = {
   249→        assetId: Number(values.assetId),
   250→        type: values.type,
   251→        quantity: Number(values.quantity),
   252→        price: Number(values.price),
   253→        fee: values.fee === "" ? 0 : Number(values.fee),
   254→        note: values.note.trim(),
   255→      };
   256→      if (values.at) {
   257→        const d = new Date(values.at);
   258→        if (!Number.isNaN(d.getTime())) body.at = d.toISOString();
   259→      }
   260→      return apiFetch<Transaction>("/transactions", { method: "POST", body });
   261→    },
   262→    onSuccess: (tx) => {
   263→      toast.success("Transaksi berhasil dicatat", {
   264→        description: `${txMeta(tx.type).label} · ${tx.assetName ?? "aset"} · ${formatNumber(tx.quantity, 6)}`,
   265→      });
   266→      void queryClient.invalidateQueries({ queryKey: ["transactions"] });
   267→      void queryClient.invalidateQueries({ queryKey: ["assets"] });
   268→      void queryClient.invalidateQueries({ queryKey: ["dashboard"] });
   269→      onOpenChange(false);
   270→    },
   271→    onError: (err) => {
   272→      toast.error("Gagal mencatat transaksi", { description: errMsg(err) });
   273→    },
   274→  });
   275→
   276→  const onSubmit = form.handleSubmit((values) => {
   277→    // Validasi client-side terkait tipe & ketersediaan aset
   278→    const t = values.type as TransactionType;
   279→    const qty = Number(values.quantity);
   280→    if (["buy", "sell", "transfer_in", "transfer_out"].includes(t) && qty <= 0) {
   281→      form.setError("quantity", { message: "Jumlah harus lebih dari 0 untuk jenis transaksi ini" });
   282→      return;
   283→    }
   284→    if (t === "sell" || t === "transfer_out") {
   285→      const asset = assets.find((a) => String(a.id) === values.assetId);
   286→      if (asset && qty > asset.quantity + 1e-9) {
   287→        form.setError("quantity", {
   288→          message: `Melebihi jumlah tersedia (${formatNumber(asset.quantity, 6)})`,
   289→        });
   290→        return;
   291→      }
   292→    }
   293→    createMutation.mutate(values);
   294→  });
   295→
   296→  const needsPositiveQty =
   297→    type === "buy" || type === "sell" || type === "transfer_in" || type === "transfer_out";
   298→
   299→  return (
   300→    <Dialog open={open} onOpenChange={onOpenChange}>
   301→      <DialogContent className="max-h-[90vh] overflow-y-auto scrollbar-thin sm:max-w-lg">
   302→        <DialogHeader>
   303→          <DialogTitle>Catat Transaksi</DialogTitle>
   304→          <DialogDescription>
   305→            Beli/jual otomatis menyesuaikan jumlah aset. Harga &amp; fee mengikuti mata uang aset.
   306→          </DialogDescription>
   307→        </DialogHeader>
   308→
   309→        <form onSubmit={onSubmit} className="grid gap-4" noValidate>
   310→          <div className="grid gap-1.5">
   311→            <Label>Aset *</Label>
   312→            <Controller
   313→              control={form.control}
   314→              name="assetId"
   315→              render={({ field }) => (
   316→                <Select value={field.value} onValueChange={field.onChange}>
   317→                  <SelectTrigger className="w-full">
   318→                    <SelectValue placeholder="Pilih aset" />
   319→                  </SelectTrigger>
   320→                  <SelectContent className="max-h-64">
   321→                    {assets.length === 0 ? (
   322→                      <div className="px-3 py-2 text-sm text-muted-foreground">
   323→                        Belum ada aset — tambahkan aset terlebih dahulu
   324→                      </div>
   325→                    ) : (
   326→                      assets.map((a) => (
   327→                        <SelectItem key={String(a.id)} value={String(a.id)}>
   328→                          {a.name}
   329→                          {a.symbol ? ` (${a.symbol})` : ""}
   330→                        </SelectItem>
   331→                      ))
   332→                    )}
   333→                  </SelectContent>
   334→                </Select>
   335→              )}
   336→            />
   337→            {selectedAsset && (
   338→              <div className="flex flex-wrap items-center gap-2 text-xs text-muted-foreground">
   339→                <ClassBadge assetClass={selectedAsset.assetClass} className="px-1.5 text-[10px]" />
   340→                <span>
   341→                  Mata uang: <span className="font-medium">{selectedAsset.currency}</span>
   342→                </span>
   343→                <span>
   344→                  Tersedia:{" "}
   345→                  <span className="font-medium tabular-nums">
   346→                    {formatNumber(selectedAsset.quantity, 6)}
   347→                    {selectedAsset.symbol ? ` ${selectedAsset.symbol}` : ""}
   348→                  </span>
   349→                </span>
   350→              </div>
   351→            )}
   352→            <FieldError message={form.formState.errors.assetId?.message} />
   353→          </div>
   354→
   355→          <div className="grid gap-1.5">
   356→            <Label>Jenis Transaksi *</Label>
   357→            <Controller
   358→              control={form.control}
   359→              name="type"
   360→              render={({ field }) => (
   361→                <Select value={field.value} onValueChange={field.onChange}>
   362→                  <SelectTrigger className="w-full">
   363→                    <SelectValue placeholder="Pilih jenis" />
   364→                  </SelectTrigger>
   365→                  <SelectContent>
   366→                    {TX_TYPE_OPTIONS.map((t) => (
   367→                      <SelectItem key={t.value} value={t.value}>
   368→                        {t.label}
   369→                      </SelectItem>
   370→                    ))}
   371→                  </SelectContent>
   372→                </Select>
   373→              )}
   374→            />
   375→            {type === "valuation_update" && (
   376→              <p className="flex items-start gap-1.5 text-xs text-muted-foreground">
   377→                <Info className="mt-0.5 h-3 w-3 shrink-0" aria-hidden="true" />
   378→                Harga baru akan diterapkan sebagai harga satuan aset.
   379→              </p>
   380→            )}
   381→            <FieldError message={form.formState.errors.type?.message} />
   382→          </div>
   383→
   384→          <div className="grid gap-4 sm:grid-cols-2">
   385→            <div className="grid gap-1.5">
   386→              <Label htmlFor="tx-quantity">Jumlah *</Label>
   387→              <Input
   388→                id="tx-quantity"
   389→                type="number"
   390→                step="any"
   391→                min="0"
   392→                inputMode="decimal"
   393→                placeholder={needsPositiveQty ? "cth. 0.25" : "cth. 0"}
   394→                {...form.register("quantity")}
   395→              />
   396→              <FieldError message={form.formState.errors.quantity?.message} />
   397→            </div>
   398→
   399→            <div className="grid gap-1.5">
   400→              <div className="flex items-center justify-between gap-2">
   401→                <Label htmlFor="tx-price">Harga Satuan *</Label>
   402→                {selectedAsset && (
   403→                  <span className="rounded-md border bg-muted px-1.5 py-0.5 text-[10px] font-semibold uppercase text-muted-foreground">
   404→                    {selectedAsset.currency}
   405→                  </span>
   406→                )}
   407→              </div>
   408→              <Input
   409→                id="tx-price"
   410→                type="number"
   411→                step="any"
   412→                min="0"
   413→                inputMode="decimal"
   414→                placeholder="cth. 65000"

   454→          <DialogFooter className="gap-2 sm:gap-0">

   490→
   491→  // ---------- queries ----------
   492→  const assetsQuery = useQuery<Asset[], ApiError>({
   493→    queryKey: ["assets", "options"],
   494→    queryFn: () => apiFetch<Asset[]>("/assets"),
   495→    staleTime: 30_000,
   496→  });
   497→  const assets = assetsQuery.data ?? [];
   498→
   499→  const txParams = {
   500→    assetId: assetFilter !== "all" ? assetFilter : undefined,
   501→    type: typeFilter !== "all" ? typeFilter : undefined,
   502→    limit: 200,
   503→  };
   504→  const txQuery = useQuery<Transaction[], ApiError>({
   505→    queryKey: ["transactions", { assetId: txParams.assetId ?? null, type: txParams.type ?? null }],
   506→    queryFn: () =>
   507→      apiFetch<Transaction[]>("/transactions", {
   508→        params: { assetId: txParams.assetId, type: txParams.type, limit: 200 },
   509→      }),
   510→    staleTime: 30_000,
   511→  });
   512→
   513→  const txs = React.useMemo(() => {
   514→    const rows = txQuery.data ?? [];
   515→    return [...rows].sort((a, b) => {
   516→      const d = (b.at ?? "").localeCompare(a.at ?? "");
   517→      return d !== 0 ? d : String(b.id).localeCompare(String(a.id));
   518→    });
   519→  }, [txQuery.data]);
   520→
   521→  const stats = React.useMemo(() => {
   522→    let inflow = 0;
   523→    let outflow = 0;
   524→    for (const tx of txs) {
   525→      if (INFLOW_TYPES.includes(tx.type)) inflow += Math.abs(tx.totalUsd ?? 0);
   526→      else if (OUTFLOW_TYPES.includes(tx.type)) outflow += Math.abs(tx.totalUsd ?? 0);
   527→    }
   528→    return { inflow, outflow, count: txs.length };
   529→  }, [txs]);
   530→
   531→  return (
   532→    <div className="space-y-6">
   533→      <PageHeader
   534→        title="Transaksi"
   535→        subtitle="Riwayat buy/sell, transfer, income/expense, dan valuation update."
   536→      />
   537→
   538→      {/* ================= StatCards arus kas ================= */}
   539→      <div className="grid gap-4 sm:grid-cols-3" aria-busy={txQuery.isPending && txs.length === 0}>
   540→        <StatCard
   541→          loading={txQuery.isPending && txs.length === 0}
   542→          label="Total Masuk"
   543→          value={formatCurrency(stats.inflow)}
   544→          sub="beli · transfer masuk · pemasukan"
   545→          icon={<ArrowDownLeft className="h-5 w-5" />}
   546→        />
   547→        <StatCard
   548→          loading={txQuery.isPending && txs.length === 0}
   549→          label="Total Keluar"
   550→          value={formatCurrency(stats.outflow)}
   551→          sub="jual, transfer keluar, pengeluaran"
   552→          accent="rose"
   553→          icon={<ArrowUpRight className="h-5 w-5" />}
   554→        />
   555→        <StatCard
   556→          loading={txQuery.isPending && txs.length === 0}
   557→          label="Jumlah Transaksi"
   558→          value={String(stats.count)}
   559→          sub={`hingga ${txParams.limit} terakhir terfilter`}
   560→          accent="amber"
   561→          icon={<ArrowLeftRight className="h-5 w-5" />}
   562→        />
   563→      </div>
   564→
   565→      {/* ================= Toolbar ================= */}
   566→      <div className="flex flex-col gap-3 lg:flex-row lg:items-center lg:justify-between">
   567→        <div className="flex flex-wrap items-center gap-3">
   568→          <Select value={typeFilter} onValueChange={setTypeFilter}>
   569→            <SelectTrigger className="w-full sm:w-[190px]" aria-label="Filter jenis transaksi">
   570→              <SelectValue placeholder="Semua Jenis" />
   571→            </SelectTrigger>
   572→            <SelectContent>
   573→              <SelectItem value="all">Semua Jenis</SelectItem>
   574→              {TX_TYPE_OPTIONS.map((t) => (
   575→                <SelectItem key={t.value} value={t.value}>
   576→                  {t.label}
   577→                </SelectItem>
   578→              ))}
   579→            </SelectContent>
   580→          </Select>
   581→
   582→          <Select value={assetFilter} onValueChange={setAssetFilter}>
   583→            <SelectTrigger className="w-full sm:w-[230px]" aria-label="Filter aset">
   584→              <SelectValue placeholder="Semua Aset" />
   585→            </SelectTrigger>
   586→            <SelectContent className="max-h-64">
   587→              <SelectItem value="all">Semua Aset</SelectItem>
   588→              {assets.map((a) => (
   589→                <SelectItem key={String(a.id)} value={String(a.id)}>
   590→                  {a.name}
   591→                  {a.symbol ? ` (${a.symbol})` : ""}
   592→                </SelectItem>
   593→              ))}
   594→            </SelectContent>
   595→          </Select>
   596→        </div>
   597→
   598→        {isOwner && (
   599→          <Button
   600→            type="button"
   601→            onClick={() => setFormOpen(true)}
   602→            className="bg-emerald-600 text-white hover:bg-emerald-700"
   603→          >
   604→            <Plus className="h-4 w-4" aria-hidden="true" />
   605→            Catat Transaksi
   606→          </Button>
   607→        )}
   608→      </div>
   609→
   610→      {/* ================= Tabel Transaksi ================= */}
   611→      <SectionCard
   612→        title="Riwayat Transaksi"
   613→        description={
   614→          txQuery.isPending
   615→            ? "Memuat data…"
   616→            : `${txs.length} transaksi ditampilkan${typeFilter !== "all" || assetFilter !== "all" ? " (terfilter)" : ""}`
   617→        }
   618→        contentClassName="px-0 pb-0"
   619→      >
   620→        {txQuery.isError && txs.length === 0 ? (
   621→          <div className="px-6 pb-6">
   622→            <ErrorBox message={errMsg(txQuery.error)} onRetry={() => void txQuery.refetch()} />
   623→          </div>
   624→        ) : txQuery.isPending && txs.length === 0 ? (
   625→          <div className="px-6 pb-6">
   626→            <LoadingRows rows={6} />
   627→          </div>
   628→        ) : txs.length === 0 ? (
   629→          <div className="px-6 pb-6">
   630→            <EmptyState
   631→              icon={<ArrowLeftRight className="h-6 w-6" />}
   632→              title="Belum ada transaksi"
   633→              description={
   634→                typeFilter !== "all" || assetFilter !== "all"
   635→                  ? "Tidak ada transaksi yang cocok dengan filter. Coba ubah kriteria."
   636→                  : isOwner
   637→                    ? "Catat transaksi pertama — beli, jual, transfer, atau pencatatan nilai."
   638→                    : "Owner belum mencatat transaksi. Riwayat akan tampil di sini."
   639→              }
   640→              action={
   641→                isOwner && typeFilter === "all" && assetFilter === "all" ? (
   642→                  <Button
   643→                    type="button"
   644→                    onClick={() => setFormOpen(true)}
   645→                    className="bg-emerald-600 text-white hover:bg-emerald-700"
   646→                  >
   647→                    <Plus className="h-4 w-4" aria-hidden="true" />
   648→                    Catat Transaksi
   649→                  </Button>
   650→                ) : undefined
   651→              }
   652→            />
   653→          </div>
   654→        ) : (
   655→          <TooltipProvider delayDuration={200}>
   656→            <div className="max-h-96 overflow-auto scrollbar-thin">
   657→              <Table>
   658→                <TableHeader className="sticky top-0 z-10 bg-card shadow-[0_1px_0_0_var(--border)]">
   659→                  <TableRow className="hover:bg-transparent">
   660→                    <TableHead className="min-w-[150px]">Tanggal</TableHead>
   661→                    <TableHead className="w-[150px]">Tipe</TableHead>
   662→                    <TableHead className="min-w-[160px]">Aset</TableHead>
   663→                    <TableHead className="w-[110px] text-right">Jumlah</TableHead>
   664→                    <TableHead className="w-[110px] text-right">Harga</TableHead>
   665→                    <TableHead className="w-[130px] text-right">Total</TableHead>
   666→                    <TableHead className="w-[100px] text-right">Fee</TableHead>
   667→                    <TableHead className="min-w-[140px]">Catatan</TableHead>
   668→                  </TableRow>
   669→                </TableHeader>
   670→                <TableBody>
   671→                  {txs.map((tx) => {
   672→                    const meta = txMeta(tx.type);
   673→                    const inflow = meta.dir === "in";
   674→                    const neutral = meta.dir === "neutral";
   675→                    const DirIcon = meta.icon;
   676→                    return (
   677→                      <TableRow key={String(tx.id)}>
   678→                        <TableCell>
   679→                          <div className="flex flex-col">
   680→                            <span className="whitespace-nowrap text-sm tabular-nums">
   681→                              {formatDate(tx.at)}
   682→                            </span>
   683→                            <span className="text-xs text-muted-foreground">{timeAgo(tx.at)}</span>
   684→                          </div>
   685→                        </TableCell>
   686→                        <TableCell>
   687→                          <span className="inline-flex items-center gap-1.5">
   688→                            <DirIcon
   689→                              className={cn(
   690→                                "h-3.5 w-3.5",
   691→                                neutral && "text-zinc-500",
   692→                                !neutral && inflow && "text-emerald-600 dark:text-emerald-400",
   693→                                !neutral && !inflow && "text-rose-600 dark:text-rose-400",
   694→                              )}
   695→                              aria-hidden="true"
   696→                            />
   697→                            <TypeBadge type={tx.type} />
   698→                          </span>
   699→                        </TableCell>
   700→                        <TableCell className="font-medium">{tx.assetName ?? `#${tx.assetId}`}</TableCell>
   701→                        <TableCell className="text-right tabular-nums">
   702→                          {formatNumber(tx.quantity, 6)}
   703→                        </TableCell>
   704→                        <TableCell className="text-right">
   705→                          <MoneyCell usd={tx.priceUsd} />
   706→                        </TableCell>
   707→                        <TableCell
   708→                          className={cn(
   709→                            "text-right font-semibold tabular-nums",
   710→                            neutral && "text-muted-foreground",
   711→                            !neutral && inflow && "text-emerald-600 dark:text-emerald-400",
   712→                            !neutral && !inflow && "text-rose-600 dark:text-rose-400",
   713→                          )}
   714→                        >
   715→                          <MoneyCell
   716→                            usd={tx.totalUsd}
   717→                            sign={neutral ? undefined : inflow ? "+" : "−"}
   718→                            className={cn(
   719→                              "font-semibold",
   720→                              neutral && "text-muted-foreground",
   721→                              !neutral && inflow && "text-emerald-600 dark:text-emerald-400",
   722→                              !neutral && !inflow && "text-rose-600 dark:text-rose-400",
   723→                            )}
   724→                          />
   725→                        </TableCell>
   726→                        <TableCell className="text-right">
   727→                          {tx.feeUsd > 0 ? (
   728→                            <MoneyCell usd={tx.feeUsd} className="text-muted-foreground" />
   729→                          ) : (
   730→                            <span className="text-muted-foreground">—</span>
   731→                          )}
   732→                        </TableCell>
   733→                        <TableCell>
   734→                          {tx.note?.trim() ? (
   735→                            <Tooltip>
   736→                              <TooltipTrigger asChild>
   737→                                <span className="block max-w-[220px] cursor-default truncate text-sm text-muted-foreground">
   738→                                  {tx.note}
   739→                                </span>
   740→                              </TooltipTrigger>
   741→                              <TooltipContent side="top" className="max-w-xs">
   742→                                <p className="text-xs">{tx.note}</p>
   743→                              </TooltipContent>
   744→                            </Tooltip>
   745→                          ) : (
   746→                            <span className="text-muted-foreground">—</span>
   747→                          )}
   748→                        </TableCell>
   749→                      </TableRow>
   750→                    );
   751→                  })}
   752→                </TableBody>
   753→              </Table>
   754→            </div>
   755→          </TooltipProvider>
   756→        )}
   757→      </SectionCard>
   758→
   759→      {txs.length > 0 && (
   760→        <p className="text-xs text-muted-foreground">
   761→          Nilai ditampilkan dalam mata uang tampilan pilihan Anda. Total masuk/keluar dihitung dari
   762→          tipe: masuk (beli, transfer masuk, pemasukan) dan keluar (jual, transfer keluar, pengeluaran).
   763→        </p>
   764→      )}
   765→
   766→      {/* ================= Dialog ================= */}
   767→      {isOwner && (
   768→        <TxFormDialog open={formOpen} onOpenChange={setFormOpen} assets={assets} />
   769→      )}
   770→    </div>
   771→  );
   772→}
