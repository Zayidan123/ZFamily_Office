   104→function Sparkline({
   105→  values,
   106→  up,
   107→  className,
   108→}: {
   109→  values: number[];
   110→  up: boolean;
   111→  className?: string;
   112→}) {
   113→  const id = React.useId().replace(/[:]/g, "");
   114→  if (values.length < 2) return null;
   115→  const w = 120;
   116→  const h = 36;
   117→  const pad = 2;
   118→  const min = Math.min(...values);
   119→  const max = Math.max(...values);
   120→  const span = max - min || 1;
   121→  const step = (w - pad * 2) / (values.length - 1);
   122→  const coords = values.map((p, i) => {
   123→    const x = pad + i * step;
   124→    const y = h - pad - ((p - min) / span) * (h - pad * 2);
   125→    return [x, y] as const;
   126→  });
   127→  const linePath = coords
   128→    .map(([x, y], i) => `${i === 0 ? "M" : "L"}${x.toFixed(1)},${y.toFixed(1)}`)
   129→    .join(" ");
   130→  const last = coords[coords.length - 1];
   131→  const first = coords[0];
   132→  const areaPath = `${linePath} L${last[0].toFixed(1)},${h - pad} L${first[0].toFixed(1)},${h - pad} Z`;
   133→  const pointsAttr = coords.map(([x, y]) => `${x.toFixed(1)},${y.toFixed(1)}`).join(" ");
   134→  const color = up ? "#10b981" : "#f43f5e";
   135→
   136→  return (
   137→    <svg
   138→      viewBox={`0 0 ${w} ${h}`}
   139→      preserveAspectRatio="none"
   140→      className={cn("h-9 w-full", className)}
   141→      aria-hidden="true"
   142→    >
   143→      <defs>
   144→        <linearGradient id={`spark-${id}`} x1="0" y1="0" x2="0" y2="1">
   145→          <stop offset="0%" stopColor={color} stopOpacity={0.32} />
   146→          <stop offset="100%" stopColor={color} stopOpacity={0.02} />
   147→        </linearGradient>
   148→      </defs>
   149→      <path d={areaPath} fill={`url(#spark-${id})`} stroke="none" />
   150→      <polyline
   151→        points={pointsAttr}
   152→        fill="none"
   153→        stroke={color}
   154→        strokeWidth={1.8}
   155→        strokeLinejoin="round"
   156→        strokeLinecap="round"
   157→      />
   158→    </svg>
   159→  );
   160→}

   165→function useMarketHistory(symbol: string, days: number, enabled = true) {
   166→  return useQuery<MarketHistory, ApiError>({
   167→    queryKey: ["market", "history", symbol, days],
   168→    queryFn: () =>
   169→      apiFetch<MarketHistory>("/market/history", { params: { symbol, days } }),
   170→    enabled: enabled && Boolean(symbol),
   171→    staleTime: 60_000,
   172→    refetchInterval: 60_000,
   173→  });
   174→}

   600→  // ---------- Mutations ----------
   601→  const invalidateWatchlist = () =>
   602→    void queryClient.invalidateQueries({ queryKey: ["market", "watchlist"] });
   603→
   604→  const addMutation = useMutation({
   605→    mutationFn: (body: { symbol: string; note?: string }) =>
   606→      apiFetch<WatchlistItem>("/market/watchlist", { method: "POST", body }),
   607→    onSuccess: (_data, vars) => {
   608→      toast.success(`${vars.symbol} ditambahkan ke watchlist`);
   609→      setAddOpen(false);
   610→      invalidateWatchlist();
   611→    },
   612→    onError: (err) => {
   613→      toast.error("Gagal menambah simbol", {
   614→        description: err instanceof ApiError ? err.message : "Coba lagi sebentar",
   615→      });
   616→    },
   617→  });
   618→
   619→  const deleteMutation = useMutation({
   620→    mutationFn: (id: number) =>
   621→      apiFetch<void>(`/market/watchlist/${id}`, { method: "DELETE" }),
   622→    onSuccess: (_d, id) => {
   623→      const target = watchlist.find((w) => w.id === id);
   624→      toast.success(`${target?.symbol ?? "Simbol"} dihapus dari watchlist`);
   625→      if (selected?.id === id) {
   626→        setDetailOpen(false);
   627→        setSelected(null);
   628→      }
   629→      invalidateWatchlist();
   630→    },
   631→    onError: (err) => {
   632→      toast.error("Gagal menghapus simbol", {
   633→        description: err instanceof ApiError ? err.message : "Coba lagi sebentar",
   634→      });
   635→    },
   636→  });
   637→
   638→  // ---------- Derived stats ----------
   639→  const gainers = watchlist.filter((w) => w.price?.change24hPct !== undefined);
   640→  const topGainer =
   641→    gainers.length > 0
   642→      ? gainers.reduce((a, b) =>
   643→          b.price.change24hPct > a.price.change24hPct ? b : a,
   644→        )
   645→      : null;
   646→  const topLoser =
   647→    gainers.length > 0
   648→      ? gainers.reduce((a, b) =>
   649→          b.price.change24hPct < a.price.change24hPct ? b : a,
   650→        )
   651→      : null;
   652→  const xau =
   653→    ratesQuery.data?.prices?.XAU ??
   654→    (symbols.find((s) => s.symbol === "XAU")
   655→      ? {
   656→          usd: symbols.find((s) => s.symbol === "XAU")!.usd,
   657→          change24hPct: symbols.find((s) => s.symbol === "XAU")!.change24hPct,
   658→        }
   659→      : null);
   660→
   661→  const loadingAll = watchlistQuery.isPending && !watchlistQuery.data;
   662→  const errMsg =
   663→    watchlistQuery.error instanceof ApiError
   664→      ? watchlistQuery.error.message
   665→      : undefined;
   666→
   667→  return (
   668→    <div className="space-y-6">
   669→      <PageHeader
   670→        title="Pasar"
   671→        subtitle="Watchlist simbol & harga live — diperbarui otomatis tiap 30 detik."
   672→        actions={
   673→          isOwner && (
   674→            <Button
   675→              type="button"
   676→              size="sm"
   677→              className="min-h-[40px] bg-emerald-600 text-white hover:bg-emerald-700"
   678→              onClick={() => setAddOpen(true)}
   679→            >
   680→              <Plus className="h-4 w-4" aria-hidden="true" />
   681→              Pantau Simbol
   682→            </Button>
   683→          )
   684→        }
   685→      />
   686→
   687→      {watchlistQuery.isError && !watchlistQuery.data && (
   688→        <ErrorBox message={errMsg} onRetry={() => void watchlistQuery.refetch()} />
   689→      )}
   690→
   691→      {/* ---------- StatCards ---------- */}
   692→      <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-4">
   693→        <StatCard
   694→          loading={loadingAll}
   695→          label="Simbol Dipantau"
   696→          value={watchlistQuery.data ? `${watchlist.length}` : "—"}
   697→          sub={`${symbols.length || "—"} simbol tersedia`}
   698→          icon={<Eye className="h-5 w-5" />}
   699→        />
   700→        <StatCard
   701→          loading={loadingAll}
   702→          label="Gainer Teratas"
   703→          value={topGainer ? topGainer.symbol : "—"}
   704→          delta={topGainer?.price.change24hPct ?? null}
   705→          sub={topGainer ? <CurrencyText usd={topGainer.price.usd} compact /> : undefined}
   706→          icon={<TrendingUp className="h-5 w-5" />}
   707→        />
   708→        <StatCard
   709→          loading={loadingAll}
   710→          label="Loser Teratas"
   711→          value={topLoser ? topLoser.symbol : "—"}
   712→          delta={topLoser?.price.change24hPct ?? null}
   713→          sub={topLoser ? <CurrencyText usd={topLoser.price.usd} compact /> : undefined}
   714→          accent="rose"
   715→          icon={<TrendingDown className="h-5 w-5" />}
   716→        />
   717→        <StatCard
   718→          label="XAU / USD"
   719→          value={xau ? formatXauValue(xau.usd) : "—"}
   720→          delta={xau?.change24hPct ?? null}
   721→          sub="Emas spot per troy ounce"
   722→          accent="amber"
   723→          icon={<Coins className="h-5 w-5" />}
   724→        />
   725→      </div>
   726→
   727→      {/* ---------- Grid watchlist ---------- */}
   728→      {loadingAll ? (
   729→        <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3" aria-busy="true">
   730→          {Array.from({ length: 3 }).map((_, i) => (
   731→            <Card key={i} className="rounded-xl p-5">
   732→              <Skeleton className="h-6 w-24" />
   733→              <Skeleton className="mt-3 h-4 w-32" />
   734→              <Skeleton className="mt-4 h-7 w-28" />
   735→              <Skeleton className="mt-4 h-9 w-full rounded-md" />
   736→            </Card>
   737→          ))}
   738→        </div>
   739→      ) : watchlist.length === 0 ? (
   740→        <EmptyState
   741→          icon={<LineChartIcon className="h-6 w-6" />}
   742→          title="Watchlist masih kosong"
   743→          description={
   744→            isOwner
   745→              ? "Pantau simbol favorit — BTC, ETH, XAU, saham, dan lainnya — lengkap dengan harga live dan sparkline 30 hari."
   746→              : "Belum ada simbol yang dipantau. Owner dapat menambahkan simbol untuk dipantau."
   747→          }
   748→          action={
   749→            isOwner && (
   750→              <Button
   751→                type="button"
   752→                size="sm"
   753→                className="bg-emerald-600 text-white hover:bg-emerald-700"
   754→                onClick={() => setAddOpen(true)}
   755→              >
   756→                <Plus className="h-4 w-4" aria-hidden="true" />
   757→                Pantau Simbol
   758→              </Button>
   759→            )
   760→          }
   761→        />
   762→      ) : (
   763→        <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3">
   764→          {watchlist.map((item) => (
   765→            <WatchCard
   766→              key={item.id}
   767→              item={item}
   768→              isOwner={isOwner}
   769→              onOpen={() => {
   770→                setSelected(item);
   771→                setDetailOpen(true);
   772→              }}
   773→              onDelete={() => deleteMutation.mutate(item.id)}
   774→              deleting={deleteMutation.isPending && deleteMutation.variables === item.id}
   775→            />
   776→          ))}
   777→        </div>
   778→      )}
   779→
   780→      {/* ---------- Dialogs ---------- */}
   781→      <WatchDetailDialog
   782→        item={selected}
   783→        open={detailOpen}
   784→        onOpenChange={setDetailOpen}
   785→      />
   786→
   787→      <AddWatchDialog
   788→        open={addOpen}
   789→        onOpenChange={setAddOpen}
   790→        symbols={symbols}
   791→        watchedSymbols={watchlist.map((w) => w.symbol)}
   792→        pending={addMutation.isPending}
   793→        onSubmit={(symbol, note) =>
   794→          addMutation.mutate({ symbol, note: note || undefined })
   795→        }
   796→      />
   797→    </div>
   798→  );
   799→}
   800→
   801→function formatXauValue(usd: number): string {
   802→  return new Intl.NumberFormat("en-US", {
   803→    style: "currency",
   804→    currency: "USD",
   805→    maximumFractionDigits: 0,
   806→  }).format(usd);
   807→}
