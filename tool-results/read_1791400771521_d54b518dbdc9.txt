   104→   104→function Sparkline({
   105→   105→  values,
   106→   106→  up,
   107→   107→  className,
   108→   108→}: {
   109→   109→  values: number[];
   110→   110→  up: boolean;
   111→   111→  className?: string;
   112→   112→}) {
   113→   113→  const id = React.useId().replace(/[:]/g, "");
   114→   114→  if (values.length < 2) return null;
   115→   115→  const w = 120;
   116→   116→  const h = 36;
   117→   117→  const pad = 2;
   118→   118→  const min = Math.min(...values);
   119→   119→  const max = Math.max(...values);
   120→   120→  const span = max - min || 1;
   121→   121→  const step = (w - pad * 2) / (values.length - 1);
   122→   122→  const coords = values.map((p, i) => {
   123→   123→    const x = pad + i * step;
   124→   124→    const y = h - pad - ((p - min) / span) * (h - pad * 2);
   125→   125→    return [x, y] as const;
   126→   126→  });
   127→   127→  const linePath = coords
   128→   128→    .map(([x, y], i) => `${i === 0 ? "M" : "L"}${x.toFixed(1)},${y.toFixed(1)}`)
   129→   129→    .join(" ");
   130→   130→  const last = coords[coords.length - 1];
   131→   131→  const first = coords[0];
   132→   132→  const areaPath = `${linePath} L${last[0].toFixed(1)},${h - pad} L${first[0].toFixed(1)},${h - pad} Z`;
   133→   133→  const pointsAttr = coords.map(([x, y]) => `${x.toFixed(1)},${y.toFixed(1)}`).join(" ");
   134→   134→  const color = up ? "#10b981" : "#f43f5e";
   135→   135→
   136→   136→  return (
   137→   137→    <svg
   138→   138→      viewBox={`0 0 ${w} ${h}`}
   139→   139→      preserveAspectRatio="none"
   140→   140→      className={cn("h-9 w-full", className)}
   141→   141→      aria-hidden="true"
   142→   142→    >
   143→   143→      <defs>
   144→   144→        <linearGradient id={`spark-${id}`} x1="0" y1="0" x2="0" y2="1">
   145→   145→          <stop offset="0%" stopColor={color} stopOpacity={0.32} />
   146→   146→          <stop offset="100%" stopColor={color} stopOpacity={0.02} />
   147→   147→        </linearGradient>
   148→   148→      </defs>
   149→   149→      <path d={areaPath} fill={`url(#spark-${id})`} stroke="none" />
   150→   150→      <polyline
   151→   151→        points={pointsAttr}
   152→   152→        fill="none"
   153→   153→        stroke={color}
   154→   154→        strokeWidth={1.8}
   155→   155→        strokeLinejoin="round"
   156→   156→        strokeLinecap="round"
   157→   157→      />
   158→   158→    </svg>
   159→   159→  );
   160→   160→}

   165→   165→function useMarketHistory(symbol: string, days: number, enabled = true) {
   166→   166→  return useQuery<MarketHistory, ApiError>({
   167→   167→    queryKey: ["market", "history", symbol, days],
   168→   168→    queryFn: () =>
   169→   169→      apiFetch<MarketHistory>("/market/history", { params: { symbol, days } }),
   170→   170→    enabled: enabled && Boolean(symbol),
   171→   171→    staleTime: 60_000,
   172→   172→    refetchInterval: 60_000,
   173→   173→  });
   174→   174→}

   563→   563→        </DialogFooter>

   566→   566→  );

   572→   572→export default function MarketView() {
   573→   573→  const queryClient = useQueryClient();
   574→   574→  const user = useAppStore((s) => s.user);
   575→   575→  const isOwner = user?.role === "owner";
   576→   576→
   577→   577→  const [addOpen, setAddOpen] = React.useState(false);
   578→   578→  const [detailOpen, setDetailOpen] = React.useState(false);
   579→   579→  const [selected, setSelected] = React.useState<WatchlistItem | null>(null);
   580→   580→
   581→   581→  // ---------- Queries ----------
   582→   582→  const watchlistQuery = useQuery<WatchlistItem[], ApiError>({
   583→   583→    queryKey: ["market", "watchlist"],
   584→   584→    queryFn: () => apiFetch<WatchlistItem[]>("/market/watchlist"),
   585→   585→    refetchInterval: 30_000,
   586→   586→    staleTime: 25_000,
   587→   587→  });
   588→   588→
   589→   589→  const symbolsQuery = useQuery<SymbolInfo[], ApiError>({
   590→   590→    queryKey: ["market", "symbols"],
   591→   591→    queryFn: () => apiFetch<SymbolInfo[]>("/market/symbols"),
   592→   592→    staleTime: 60_000,
   593→   593→  });
   594→   594→
   595→   595→  const ratesQuery = useRates();
   596→   596→
   597→   597→  const watchlist = watchlistQuery.data ?? [];
   598→   598→  const symbols = symbolsQuery.data ?? [];
   599→   599→
   600→   600→  // ---------- Mutations ----------
   601→   601→  const invalidateWatchlist = () =>
   602→   602→    void queryClient.invalidateQueries({ queryKey: ["market", "watchlist"] });
   603→   603→
   604→   604→  const addMutation = useMutation({
   605→   605→    mutationFn: (body: { symbol: string; note?: string }) =>
   606→   606→      apiFetch<WatchlistItem>("/market/watchlist", { method: "POST", body }),
   607→   607→    onSuccess: (_data, vars) => {
   608→   608→      toast.success(`${vars.symbol} ditambahkan ke watchlist`);
   609→   609→      setAddOpen(false);
   610→   610→      invalidateWatchlist();
   611→   611→    },
   612→   612→    onError: (err) => {
   613→   613→      toast.error("Gagal menambah simbol", {
   614→   614→        description: err instanceof ApiError ? err.message : "Coba lagi sebentar",
   615→   615→      });
   616→   616→    },
   617→   617→  });
   618→   618→
   619→   619→  const deleteMutation = useMutation({
   620→   620→    mutationFn: (id: number) =>
   621→   621→      apiFetch<void>(`/market/watchlist/${id}`, { method: "DELETE" }),
   622→   622→    onSuccess: (_d, id) => {
   623→   623→      const target = watchlist.find((w) => w.id === id);
   624→   624→      toast.success(`${target?.symbol ?? "Simbol"} dihapus dari watchlist`);
   625→   625→      if (selected?.id === id) {
   626→   626→        setDetailOpen(false);
   627→   627→        setSelected(null);
   628→   628→      }
   629→   629→      invalidateWatchlist();
   630→   630→    },
   631→   631→    onError: (err) => {
   632→   632→      toast.error("Gagal menghapus simbol", {
   633→   633→        description: err instanceof ApiError ? err.message : "Coba lagi sebentar",
   634→   634→      });
   635→   635→    },
   636→   636→  });
   637→   637→
   638→   638→  // ---------- Derived stats ----------
   639→   639→  const gainers = watchlist.filter((w) => w.price?.change24hPct !== undefined);
   640→   640→  const topGainer =
   641→   641→    gainers.length > 0
   642→   642→      ? gainers.reduce((a, b) =>
   643→   643→          b.price.change24hPct > a.price.change24hPct ? b : a,
   644→   644→        )
   645→   645→      : null;
   646→   646→  const topLoser =
   647→   647→    gainers.length > 0
   648→   648→      ? gainers.reduce((a, b) =>
   649→   649→          b.price.change24hPct < a.price.change24hPct ? b : a,
   650→   650→        )
   651→   651→      : null;
   652→   652→  const xau =
   653→   653→    ratesQuery.data?.prices?.XAU ??
   654→   654→    (symbols.find((s) => s.symbol === "XAU")
   655→   655→      ? {
   656→   656→          usd: symbols.find((s) => s.symbol === "XAU")!.usd,
   657→   657→          change24hPct: symbols.find((s) => s.symbol === "XAU")!.change24hPct,
   658→   658→        }
   659→   659→      : null);
   660→   660→
   661→   661→  const loadingAll = watchlistQuery.isPending && !watchlistQuery.data;
   662→   662→  const errMsg =
   663→   663→    watchlistQuery.error instanceof ApiError
   664→   664→      ? watchlistQuery.error.message
   665→   665→      : undefined;
   666→   666→
   667→   667→  return (
   668→   668→    <div className="space-y-6">
   669→   669→      <PageHeader

   672→   672→        actions={
   673→   673→          isOwner && (
   674→   674→            <Button
   675→   675→              type="button"
   676→   676→              size="sm"
   677→   677→              className="min-h-[40px] bg-emerald-600 text-white hover:bg-emerald-700"
   678→   678→              onClick={() => setAddOpen(true)}
   679→   679→            >
   680→   680→              <Plus className="h-4 w-4" aria-hidden="true" />
   681→   681→              Pantau Simbol
   682→   682→            </Button>
   683→   683→          )
   684→   684→        }
   685→   685→      />
   686→   686→
   687→   687→      {watchlistQuery.isError && !watchlistQuery.data && (
   688→   688→        <ErrorBox message={errMsg} onRetry={() => void watchlistQuery.refetch()} />
   689→   689→      )}
   690→   690→
   691→   691→      {/* ---------- StatCards ---------- */}
   692→   692→      <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-4">
   693→   693→        <StatCard
   694→   694→          loading={loadingAll}
   695→   695→          label="Simbol Dipantau"
   696→   696→          value={watchlistQuery.data ? `${watchlist.length}` : "—"}
   697→   697→          sub={`${symbols.length || "—"} simbol tersedia`}

   726→   726→
   727→   727→      {/* ---------- Grid watchlist ---------- */}
   728→   728→      {loadingAll ? (

   730→   730→          {Array.from({ length: 3 }).map((_, i) => (
